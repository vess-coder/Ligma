-- ==========================================================================
-- Connection scopes: every module owns a scope so disabling a feature
-- disconnects exactly what it connected (no duplicates, no leaks).
-- ==========================================================================
local function newScope()
    local scope = { _list = {} }
    function scope.connect(signal, fn)
        local c = signal:Connect(fn)
        scope._list[#scope._list + 1] = c
        return c
    end
    function scope.add(handle)
        -- Accepts RBXScriptConnections and threads (from task.spawn) alike.
        if handle ~= nil then
            scope._list[#scope._list + 1] = handle
        end
        return handle
    end
    function scope.disconnectAll()
        for _, handle in ipairs(scope._list) do
            if type(handle) == "thread" then
                pcall(function() task.cancel(handle) end)
            else
                pcall(function() handle:Disconnect() end)
            end
        end
        scope._list = {}
    end
    return scope
end

-- Global scope: lives for the whole session, cleared on unload.

function Hub.applyLayout(name)
    local preset = LayoutPresets[name] or LayoutPresets.PC
    Hub.Layout = LayoutPresets[name] and name or "PC"

    local viewport = Workspace.CurrentCamera and Workspace.CurrentCamera.ViewportSize or Vector2.new(1280, 720)
    local scale = 1
    if UserInputService.TouchEnabled and not UserInputService.MouseEnabled then
        -- Mobile scale derived the same way the reference computes its UI scale.
        local compact = math.clamp(math.min(viewport.X, viewport.Y) / 820, 0.62, 0.92)
        scale = compact
    end
    local width = math.floor(preset.size.X * scale + 0.5)
    local height = math.floor(preset.size.Y * scale + 0.5)
    -- Never exceed 92% of the viewport on either axis.
    width = math.min(width, math.floor(viewport.X * 0.92))
    height = math.min(height, math.floor(viewport.Y * 0.86))

    Hub.Root.Size = UDim2.new(0, width, 0, height)
    Hub.Root.Position = UDim2.new(0.5, -math.floor(width / 2), 0.5, -math.floor(height / 2))
    Hub.HeaderTitle.TextSize = preset.titleSize + 2
    Hub.HeaderSub.TextSize = preset.titleSize - 4
    for _, tab in pairs(Hub.Tabs) do
        tab.button.Label.TextSize = preset.controlSize
        tab.button.Icon.TextSize = preset.controlSize
    end
    Hub.TargetGuiName = preset.gui
    if HubContext.onLayoutChanged then pcall(HubContext.onLayoutChanged) end
end

function Hub.setAccent(color)
    Hub.Accent = color
    Palette.Accent = color
    Palette.AccentDim = Color3.new(color.R * 0.72, color.G * 0.72, color.B * 0.72)
    for _, entry in ipairs(Hub.AccentTargets or {}) do
        pcall(function()
            if entry.Property == "Color" then
                entry.Object.BackgroundColor3 = color
            elseif entry.Property == "Stroke" then
                entry.Object.Color = color
            elseif entry.Property == "Text" then
                entry.Object.TextColor3 = color
            end
        end)
    end
end

    Hub.AccentTargets = Hub.AccentTargets or {}

end

    -- Slider ----------------------------------------------------------------
    function section.AddSlider(config)
        local min = tonumber(config.Min) or 0
        local max = tonumber(config.Max) or 100
        local increment = tonumber(config.Increment) or 1
        local value = tonumber(config.Default) or min
        local suffix = config.Suffix and (" " .. config.Suffix) or ""

        local row = nextRow(46)
        local label = Instance.new("TextLabel")
        label.Name = "Label"
        label.BackgroundTransparency = 1
        label.Position = UDim2.new(0, 10, 0, 5)

        label.Font = FONT_MED

        label.TextXAlignment = Enum.TextXAlignment.Left
        label.TextColor3 = Palette.TextPrimary
        label.Text = tostring(config.Text or "Slider")
        label.Parent = row

        local valueLabel = Instance.new("TextLabel")
        valueLabel.Name = "Value"
        valueLabel.BackgroundTransparency = 1
        valueLabel.AnchorPoint = Vector2.new(1, 0)
        valueLabel.Position = UDim2.new(1, -10, 0, 5)
        valueLabel.Size = UDim2.new(0, 80, 0, 14)
        valueLabel.Font = FONT_MED
        valueLabel.TextSize = 12
        valueLabel.TextXAlignment = Enum.TextXAlignment.Right
        valueLabel.TextColor3 = Hub.Accent
        valueLabel.Text = tostring(value) .. suffix
        valueLabel.Parent = row
        trackAccent(valueLabel, "Text")

        local bar = Instance.new("Frame")
        bar.Name = "Bar"
        bar.Position = UDim2.new(0, 10, 0, 26)
        bar.Size = UDim2.new(1, -20, 0, 6)
        bar.BackgroundColor3 = Palette.Input
        bar.Active = true
        bar.Parent = row
        corner(bar, 3)

        local fill = Instance.new("Frame")
        fill.Name = "Fill"
        fill.Size = UDim2.new(0, 0, 1, 0)
        fill.BackgroundColor3 = Hub.Accent
        fill.BorderSizePixel = 0
        fill.Parent = bar
        corner(fill, 3)
        trackAccent(fill, "Color")

        local function quantize(v)
            v = math.clamp(v, min, max)
            v = math.floor(v / increment + 0.5) * increment
            -- Trim floating point noise from increments like 0.1
            v = math.floor(v * 1000 + 0.5) / 1000
            return math.clamp(v, min, max)
        end

        local function render()
            local alpha = (value - min) / math.max(max - min, 1e-6)
            fill.Size = UDim2.new(alpha, 0, 1, 0)
            valueLabel.Text = tostring(value) .. suffix
        end
        render()

        local callbacks = { config.Callback }
        local function commit(v, silent)
            value = quantize(v)
            render()
            if not silent then
                for _, cb in ipairs(callbacks) do
                    if cb then pcall(cb, value) end
                end
                if HubContext.onControlChanged and config.Flag then
                    pcall(HubContext.onControlChanged, config.Flag, value)
                end
            end
        end

        local dragging = false
        bar.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                local alpha = math.clamp((input.Position.X - bar.AbsolutePosition.X) / math.max(bar.AbsoluteSize.X, 1), 0, 1)
                commit(min + (max - min) * alpha)
            end
        end)
        bar.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = false
            end
        end)
        GlobalScope.add(UserInputService.InputChanged:Connect(function(input)
            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                local alpha = math.clamp((input.Position.X - bar.AbsolutePosition.X) / math.max(bar.AbsoluteSize.X, 1), 0, 1)
                commit(min + (max - min) * alpha)
            end
        end))

        return {
            Get = function() return value end,
            Set = function(v) commit(v, true) end,
            OnChange = function(cb) callbacks[#callbacks + 1] = cb end,
        }
    end

local function normalize(s)
    s = string.lower(s or "")
    s = s:gsub("’", "'"):gsub("‘", "'")
    return (s:gsub("[%p%c%s]", ""))
end

local function isChromaName(value)
    local raw = string.lower(tostring(value or "")):gsub("^%s+", ""):gsub("%s+$", "")
    local compact = normalize(raw)
    -- "Chromatic" is an ordinary item name, not a Chroma modifier.
    if compact == "chromatic" or string.sub(compact, 1, 9) == "chromatic" then return false end
    -- A game catalogue key can contain a hierarchy, e.g.
    -- "ItemModule.Chroma.Darkbringer". Detect Chroma as a complete path/name
    -- segment as well as the compact display forms "ChromaDarkbringer" and
    -- "DarkbringerChroma"; do not confuse it with "Chromatic".
    local searchAt = 1
    while true do
        local startAt, endAt = string.find(raw, "chroma", searchAt, true)
        if not startAt then break end
        local before = startAt == 1 and "" or string.sub(raw, startAt - 1, startAt - 1)
        local after = string.sub(raw, endAt + 1, endAt + 1)
        local leftBoundary = before == "" or not string.match(before, "[%w]")
        local rightBoundary = after == "" or not string.match(after, "[%w]")
        if leftBoundary and rightBoundary then return true end
        searchAt = endAt + 1
    end
    return string.match(raw, "^c%.%s*") ~= nil
        or string.sub(compact, 1, 6) == "chroma"
        or string.sub(compact, -6) == "chroma"
end

local function stripHtml(html)
    html = html:gsub("–", "-"):gsub("—", "-"):gsub("−", "-"):gsub("‑", "-")
    html = html:gsub("<script[%s%S]-</script>", " ")
    html = html:gsub("<style[%s%S]-</style>", " ")
    html = html:gsub("<[^>]->", " ")
    html = html:gsub("&nbsp;", " ")
    html = html:gsub("&amp;", "&")
    html = html:gsub("&quot;", '"')
    html = html:gsub("&#39;", "'")
    html = html:gsub("&#039;", "'")
    html = html:gsub("&apos;", "'")
    html = html:gsub("&rsquo;", "'")
    html = html:gsub("&lsquo;", "'")
    html = html:gsub("&#x27;", "'")
    html = html:gsub("&#8217;", "'")
    html = html:gsub("&ndash;", "-")
    html = html:gsub("&mdash;", "-")
    html = html:gsub("&#8211;", "-")
    html = html:gsub("&#8212;", "-")
    html = html:gsub("%s+", " ")
    return html
end

local Database = {}
local isDatabaseLoaded = false
local loadedPagesCount = 0
local totalPagesCount = #Pages
local ValuesStatusText = "Loading local values..."
local ValuesItemCount = 0

local function parsePage(html, database, category)
    local text = stripHtml(html)
    local pos = 1
    local count = 0

    local function parsedName(rawName)
        local pipePos = rawName:match(".*()|")
        local name = pipePos and rawName:sub(pipePos + 1) or rawName
        name = name:match("^(.-)%s+Contains%s*[-:]%s*") or name
        name = name:gsub("^.*Tier%s+%d+%s+", "")
        name = name:gsub("%s+", " "):gsub("^%s+", ""):gsub("%s+$", "")
        if category == "chroma" then
            name = name:gsub("^[Cc]%s*%.%s*", "Chroma ")
            name = name:gsub("^[Cc]%s+", "Chroma ")
        elseif category == "unique" and string.find(normalize(name), "corrupt", 1, true) then
            name = "Corrupt"
        end
        return name
    end

    local function metadataAfter(valueEnd)
        -- Each Supreme card's text continues through its Origin and ends
        -- before the next Value row (or at the Change in Value field).
        local nextValueAt = string.find(text, "Value%s*[-:]%s*", valueEnd + 1)
        local finish = nextValueAt and (nextValueAt - 1) or math.min(#text, valueEnd + 500)
        local tail = text:sub(valueEnd + 1, finish)
        local origin = tail:match("Origin%s*[-:]%s*(.-)%s*$")
        if origin then origin = origin:gsub("%s+Inv%.?%s*Controls.*$", "") end
        local year = origin and origin:match("(20%d%d)") or nil
        local stability = tail:match("Stability%s*[-:]%s*(.-)%s+Demand")
        return stability and stability:gsub("%s+$", "") or "Stable", origin, year
    end

    while true do
        local startAt, endAt, rawName, rawValue = string.find(
            text,
            "([%w][%w%s'’|,.%-()]-%s*)Value%s*[-:]%s*([%d,%.]+)",
            pos
        )

        if not startAt then break end

        local name = parsedName(rawName)
        local stability, origin, year = metadataAfter(endAt)

        local value = tonumber((rawValue:gsub(",", "")))
        if name ~= "" and value and value < 100000 then
            local key = normalize(name)
            local itemData = {
                Value = value,
                Stability = stability,
                Category = category,
                Origin = origin,
                Year = year,
                ValueText = rawValue
            }

            database[key] = database[key] or {}
            table.insert(database[key], itemData)

            count = count + 1
        end

        pos = endAt + 1
    end

    -- Keep tier-based Supreme entries in the catalogue as non-numeric rows.
    -- Otherwise a same-name numeric variant (e.g. Cane Knife 2018) could be
    -- incorrectly used for an older Cane whose Supreme value is xN T1 items.
    pos = 1
    while true do
        local startAt, endAt, rawName, rawValue = string.find(
            text,
            "([%w][%w%s'’|,.%-()]-%s*)Value%s*[-:]%s*(x%s*%d+%s+T%s*%d+%s+[%w]+)",
            pos
        )
        if not startAt then break end
        local name = parsedName(rawName)
        local stability, origin, year = metadataAfter(endAt)
        if name ~= "" then
            local key = normalize(name)
            database[key] = database[key] or {}
            table.insert(database[key], {
                Value = 0,
                ValueText = rawValue,
                Stability = stability,
                Category = category,
                Origin = origin,
                Year = year
            })
            count = count + 1
        end
        pos = endAt + 1
    end

    -- Supreme Values has used a different display label for this row across
    -- page revisions ("C. Darkbringer" / "Chroma Darkbringer"). If the general
    -- row parser misses it, recover the live value from the same Chroma page
    -- and store it under the canonical key used by GetItemData.
    if category == "chroma" then
        local hasDarkbringerChroma = false
        for _, entry in ipairs(database["chromadarkbringer"] or {}) do
            if string.lower(tostring(entry.Category or "")) == "chroma" then
                hasDarkbringerChroma = true
                break
            end
        end
        if not hasDarkbringerChroma then
            local _, endAt, rawValue = string.find(text, "[Dd]arkbringer%s+Value%s*[-:]%s*([%d,]+)")
            local value = rawValue and tonumber((rawValue:gsub(",", "")))
            if value and value > 0 then
                local tail = text:sub(endAt + 1, endAt + 500)
                local stability = tail:match("Stability%s*[-:]%s*(.-)%s+Demand")
                database["chromadarkbringer"] = database["chromadarkbringer"] or {}
                table.insert(database["chromadarkbringer"], {
                    Value = value,
                    Stability = stability and stability:gsub("%s+$", "") or "Stable",
                    Category = "chroma"
                })
                count = count + 1
            end
        end
    end
    return count
end

local function fetchPage(url)
    local ok, result = pcall(function()
        return game:HttpGet(url)
    end)
    if not ok then return nil end
    if type(result) ~= "string" or #result < 100 then return nil end
    return result
end

local function getSupremeVersion(html)
    if type(html) ~= "string" then return nil end
    local text = stripHtml(html)
    return text:match("Values Last Updated%s*[-:]%s*(.-)%s*//")
        or text:match("Values Last Updated%s*[-:]%s*([^%.]+)")
end

local function readValueCache()
    if type(ExecutorEnv.isfile) ~= "function" or type(ExecutorEnv.readfile) ~= "function" then return nil end
    local existsOk, exists = pcall(ExecutorEnv.isfile, VALUE_CACHE_FILE)
    if not existsOk or not exists then return nil end
    local readOk, contents = pcall(ExecutorEnv.readfile, VALUE_CACHE_FILE)
    if not readOk or type(contents) ~= "string" or contents == "" then return nil end
    local decodeOk, cached = pcall(function()
        return HttpService:JSONDecode(contents)
    end)
    if not decodeOk or type(cached) ~= "table" or type(cached.Database) ~= "table" then
        return nil
    end
    local hasRows = false
    for _, candidates in pairs(cached.Database) do
        if type(candidates) == "table" then
            if tonumber(candidates.Value) then
                hasRows = true
                break
            end
            for _, entry in ipairs(candidates) do
                if type(entry) == "table" and tonumber(entry.Value) then
                    hasRows = true
                    break
                end
            end
            if hasRows then break end
        end
    end
    if not hasRows then return nil end
    return cached
end

local function writeValueCache(sourceVersion, database)
    if type(ExecutorEnv.writefile) ~= "function" or type(sourceVersion) ~= "string" then return false end
    local encodeOk, contents = pcall(function()
        return HttpService:JSONEncode({
            Schema = 1,
            SourceVersion = sourceVersion,
            Database = database
        })
    end)
    if not encodeOk or type(contents) ~= "string" then return false end
    local writeOk = pcall(ExecutorEnv.writefile, VALUE_CACHE_FILE, contents)
    return writeOk
end

local function useCachedDatabase(cached)
    for key, entries in pairs(cached.Database) do
        Database[key] = entries
    end
    isDatabaseLoaded = true
end

local valuesRefreshBusy = false
local function refreshValuesDatabase(onStatus)
    if valuesRefreshBusy then return end
    valuesRefreshBusy = true
    local status = onStatus or function() end
    status("Checking local values...", Palette.TextSecondary)
    local cached = readValueCache()

    -- The first category page provides Supreme's global last-updated stamp.
    -- If it matches the cache, skip downloading the other categories.
    status("Checking Supreme version...", Palette.TextSecondary)
    local firstPage = Pages[1]
    local firstHtml = firstPage and fetchPage(firstPage.url) or nil
    local sourceVersion = getSupremeVersion(firstHtml)

    if cached and sourceVersion and cached.SourceVersion == sourceVersion then
        useCachedDatabase(cached)
        status("CACHED VALUES • UP TO DATE", Palette.Success)
        valuesRefreshBusy = false
        return
    end

    if not firstHtml then
        if cached then
            useCachedDatabase(cached)
            status("OFFLINE • USING LOCAL CACHE", Palette.Warning)
        else
            status("NO NETWORK OR LOCAL CACHE", Palette.Danger)
        end
        valuesRefreshBusy = false
        return
    end

    local refreshedDatabase = {}
    local successfulPages = 0
    for index, pageData in ipairs(Pages) do
        local html = index == 1 and firstHtml or fetchPage(pageData.url)
        if html and parsePage(html, refreshedDatabase, pageData.cat) > 0 then
            successfulPages = successfulPages + 1
        end
        loadedPagesCount = index
        status(string.format("UPDATING VALUES (%d/%d)", index, totalPagesCount), Palette.TextSecondary)
        task.wait()
    end

    if successfulPages == totalPagesCount then
        for key, entries in pairs(refreshedDatabase) do
            Database[key] = entries
        end
        isDatabaseLoaded = true
        local saved = sourceVersion and writeValueCache(sourceVersion, refreshedDatabase) or false
        status(saved and "VALUES UPDATED AND SAVED" or "VALUES UPDATED • CACHE UNAVAILABLE", Palette.Success)
    elseif cached then
        useCachedDatabase(cached)
        status("UPDATE FAILED • USING OLD CACHE", Palette.Warning)
    else
        status(string.format("LOAD FAILED (%d/%d)", successfulPages, totalPagesCount), Palette.Danger)
    end
    valuesRefreshBusy = false
end

        if HubContext.valuesStatus then
            pcall(function()
                HubContext.valuesStatus.Set("busy", text)
            end)
        end
    end)
end)

-- ==========================================================================
-- [S4] GAME CATALOGUE + GET ITEM DATA (preserved from NovaTradeHelper)
-- ==========================================================================
-- Rarity and Chroma come from the game catalogue and card tags, never from
-- GUI colors. The indexed catalogue lookup and every fallback below are the
-- original implementation.
local TargetGuiName = "TradeGUI"

local ItemRecordSets
local GameItemRecords
local GameRecordIndexes
local GameCatalogueBuilding = false

local function getItemRecords()
    if ItemRecordSets then return ItemRecordSets end
    ItemRecordSets = {}
    local function addRecordSet(value)
        if type(value) ~= "table" then return end
        for _, existing in ipairs(ItemRecordSets) do
            if existing == value then return end
        end
        table.insert(ItemRecordSets, value)
    end
    local database = ReplicatedStorage:FindFirstChild("Database")
    local sync = database and database:FindFirstChild("Sync")
    if sync and sync:IsA("ModuleScript") then
        -- Some game builds expose the catalogue as Database.Sync itself,
        -- with Item or Weapons nested in its returned table.
        local ok, result = pcall(require, sync)
        if ok and type(result) == "table" then
            addRecordSet(result.Item)
            addRecordSet(result.Weapons)
            addRecordSet(result)
        end
    elseif sync then
        -- Other builds expose Sync as a Folder containing Item/Weapons modules.
        for _, moduleName in ipairs({"Item", "Weapons"}) do
            local itemModule = sync:FindFirstChild(moduleName)
            if itemModule and itemModule:IsA("ModuleScript") then
                local ok, result = pcall(require, itemModule)
                if ok then addRecordSet(result) end
            end
        end
    end
    local modules = ReplicatedStorage:FindFirstChild("Modules")
    local itemModule = modules and modules:FindFirstChild("ItemModule")
    if itemModule and itemModule:IsA("ModuleScript") then
        local ok, result = pcall(require, itemModule)
        if ok then addRecordSet(result) end
    end
    return ItemRecordSets
end

local function getFrameIdentity(itemFrame)
    if not itemFrame then return nil, nil, nil end
    local itemId, itemType
    local idAttributes = {"ItemId", "ItemID", "ID", "Id", "ItemKey"}
    local typeAttributes = {"WeaponType", "ItemType", "Type"}
    local function readWeaponType(value)
        local kind = normalize(tostring(value or ""))
        if kind == "gun" or kind == "guns" then return "gun" end
        if kind == "knife" or kind == "knives" then return "knife" end
        return nil
    end
    local function readIdentity(instance)
        if not itemId then
            for _, attr in ipairs(idAttributes) do
                local ok, value = pcall(function() return instance:GetAttribute(attr) end)
                if ok and value ~= nil then itemId = tostring(value); break end
            end
            if not itemId and instance:IsA("ValueBase") then
                local name = string.lower(instance.Name)
                if name == "itemid" or name == "itemkey" or name == "id" then
                    itemId = tostring(instance.Value)
                end
            end
        end
        if not itemType then
            for _, attr in ipairs(typeAttributes) do
                local ok, value = pcall(function() return instance:GetAttribute(attr) end)
                local kind = ok and readWeaponType(value)
                if kind then itemType = kind; break end
            end
            if not itemType and instance:IsA("ValueBase") then
                local name = normalize(instance.Name)
                if name == "weapontype" or name == "itemtype" or name == "type" then
                    local ok, value = pcall(function() return instance.Value end)
                    if ok then itemType = readWeaponType(value) end
                end
            end
            if not itemType then
                local name = normalize(instance.Name)
                if string.match(name, "knife$") then itemType = "knife"
                elseif string.match(name, "gun$") then itemType = "gun" end
            end
        end
    end
    readIdentity(itemFrame)
    if not itemId or not itemType then
        for _, desc in ipairs(itemFrame:GetDescendants()) do
            readIdentity(desc)
            if itemId and itemType then break end
        end
    end
    if not itemType then
        local name = normalize(itemFrame.Name)
        if string.find(name, "knife", 1, true) then itemType = "knife"
        elseif string.find(name, "gun", 1, true) then itemType = "gun" end
    end
    local container = itemFrame:FindFirstChild("Container") or itemFrame:FindFirstChild("Container", true)
    local icon = container and container:FindFirstChild("Icon", true)
    local image = icon and (icon:IsA("ImageLabel") or icon:IsA("ImageButton")) and icon.Image or nil
    return itemId, image, itemType
end

local function getImageAssetId(value)
    if value == nil then return nil end
    local text = tostring(value)
    -- Trade icons can use rbxthumb URLs with either `id=` or `assetId=`,
    -- for example `...width=250&height=250&assetId=8304772774`. Extract the
    -- asset parameter instead of accidentally matching thumbnail dimensions.
    local id = string.match(text, "[?&]id=(%d+)")
        or string.match(text, "[?&]assetId=(%d+)")
        or string.match(text, "rbxassetid://(%d+)")
        or string.match(text, "^%s*(%d+)%s*$")
    return id
end

local function getFrameWeaponType(itemFrame)
    local _, _, itemType = getFrameIdentity(itemFrame)
    return itemType
end

local function findGameItem(itemName, itemFrame)
    local wanted = normalize(itemName)
    local wantedId, wantedImage, wantedType = getFrameIdentity(itemFrame)
    local candidates, imageMatches = {}, {}
    local wantedAsset = getImageAssetId(wantedImage)
    while GameCatalogueBuilding do task.wait() end
    if not GameItemRecords then
        GameCatalogueBuilding = true
        local recordsOk = pcall(function()
            GameItemRecords = {}
            local catalogueSteps = 0
            local function collect(node, path, visited, depth)
                if type(node) ~= "table" or visited[node] or depth > 8 then return end
                visited[node] = true
                local hasRecordField = node.Name ~= nil or node.ItemName ~= nil or node.Rarity ~= nil
                    or node.ItemRarity ~= nil or node.RarityName ~= nil or node.Category ~= nil
                    or node.ItemId ~= nil or node.ItemID ~= nil or node.Image ~= nil or node.Icon ~= nil
                    or node.TextureId ~= nil
                if hasRecordField then
                    table.insert(GameItemRecords, {Key = path, Data = node})
                end
                -- Keep descending: some ItemModule versions place a Rarity field
                -- on a category/group table that also contains the actual item
                -- records. Stopping at the group hid those child records.
                for key, value in pairs(node) do
                    if type(value) == "table" then
                        collect(value, path == "" and tostring(key) or (path .. "." .. tostring(key)), visited, depth + 1)
                    end
                    catalogueSteps = catalogueSteps + 1
                    if catalogueSteps % 400 == 0 then task.wait() end
                end
            end
            for _, recordSet in ipairs(getItemRecords()) do
                collect(recordSet, "", {}, 0)
            end
        end)
        if not recordsOk then GameItemRecords = {} end
        GameCatalogueBuilding = false
    end
    local function recordIsChroma(record, key)
        local rarity = string.lower(tostring(record.Rarity or record.ItemRarity or record.RarityName or record.Category or ""))
        return rarity == "chroma" or isChromaName(record.Name or record.ItemName) or isChromaName(key)
    end
    local function baseRecordName(value)
        local name = normalize(tostring(value or ""))
        name = name:gsub("^vintage", ""):gsub("vintage$", "")
        name = name:gsub("^classic", ""):gsub("classic$", "")
        name = name:gsub("20%d%d$", "")
        name = name:gsub("gun$", ""):gsub("knife$", "")
        return name
    end
    local function recordRarityKey(record, key)
        local rarity = string.lower(tostring(record.Rarity or record.ItemRarity or record.RarityName or record.Category or ""))
        if rarity == "classic" then return "vintage" end
        if rarity ~= "" then return rarity end
        local normalizedKey = normalize(tostring(key or ""))
        for _, candidate in ipairs({"vintage", "classic", "ancient", "godly", "legendary", "rare", "uncommon", "common", "unique", "chroma"}) do
            if candidate == "chroma" then
                if isChromaName(key) then return candidate end
            elseif string.find(normalizedKey, candidate, 1, true) then
                return candidate
            end
        end
        return ""
    end
    local function imageAsset(value, depth)
        if depth > 3 or value == nil then return nil end
        if type(value) == "number" or type(value) == "string" then
            return getImageAssetId(value)
        end
        if type(value) == "table" then
            for key, nested in pairs(value) do
                local keyName = string.lower(tostring(key))
                if string.find(keyName, "image", 1, true) or string.find(keyName, "icon", 1, true)
                    or string.find(keyName, "texture", 1, true) or string.find(keyName, "thumbnail", 1, true) then
                    local found = imageAsset(nested, depth + 1)
                    if found then return found end
                end
            end
        end
        return nil
    end
    local function canonicalRecordName(record, key)
        local name = record.Name or record.ItemName
        if not name then
            name = tostring(key or ""):match("([^%.]+)$") or tostring(key or "")
        end
        return baseRecordName(name)
    end
    local function recordType(record)
        return normalize(tostring(record.Type or record.ItemType or record.WeaponType or record.ItemClass or ""))
    end

    -- Build the game's catalogue indexes once. Previously every inventory card
    -- walked every record and re-parsed its icon/type/name; large inventories
    -- made that O(cards * catalogue) work stall low-end devices.
    if not GameRecordIndexes then
        local indexes = {ById = {}, ByImage = {}, ByName = {}}
        local function add(index, key, entry)
            if key == nil or key == "" then return end
            local bucket = index[key]
            if not bucket then bucket = {}; index[key] = bucket end
            bucket[#bucket + 1] = entry
        end
        for recordIndex, entry in ipairs(GameItemRecords) do
            local key, record = entry.Key, entry.Data
            entry.Index = recordIndex
            entry.RecordId = tostring(record.ItemId or record.ItemID or record.Id or record.ID or key)
            local recordImage = record.Image or record.ImageId or record.ImageID or record.ImageAssetId
                or record.Icon or record.IconId or record.IconImage or record.IconImageId
                or record.ItemImage or record.ImageAsset or record.Texture or record.TextureId
                or record.Thumbnail or record.ThumbnailId
            entry.RecordAsset = imageAsset(recordImage, 0)
            entry.RecordType = recordType(record)
            entry.RarityKey = recordRarityKey(record, key)
            entry.IsChroma = recordIsChroma(record, key)
            entry.CanonicalName = canonicalRecordName(record, key)

            add(indexes.ById, entry.RecordId, entry)
            add(indexes.ByImage, entry.RecordAsset, entry)

            local recordName = record.Name or record.ItemName or key
            local lastKeyPart = tostring(key or ""):match("([^%.]+)$") or tostring(key or "")
            local aliases = {
                normalize(tostring(recordName)), normalize(tostring(key)),
                baseRecordName(recordName), baseRecordName(lastKeyPart)
            }
            local added = {}
            for _, alias in ipairs(aliases) do
                if alias ~= "" and not added[alias] then
                    added[alias] = true
                    add(indexes.ByName, alias, entry)
                end
            end
        end
        GameRecordIndexes = indexes
    end

    if wantedId then
        local idMatches = GameRecordIndexes.ById[wantedId]
        if idMatches and idMatches[1] then return idMatches[1].Data, idMatches[1].Key end
    end
    if wantedAsset then
        for _, entry in ipairs(GameRecordIndexes.ByImage[wantedAsset] or {}) do
            imageMatches[#imageMatches + 1] = entry
        end
    end
    for _, entry in ipairs(GameRecordIndexes.ByName[wanted] or {}) do
        candidates[#candidates + 1] = entry
    end
    if wantedType then
        local typedImages, typedCandidates = {}, {}
        for _, entry in ipairs(imageMatches) do
            if entry.RecordType == wantedType then table.insert(typedImages, entry) end
        end
        for _, entry in ipairs(candidates) do
            if entry.RecordType == wantedType then table.insert(typedCandidates, entry) end
        end
        if #typedImages > 0 then imageMatches = typedImages end
        if #typedCandidates > 0 then candidates = typedCandidates end
    end
    if #imageMatches == 1 then return imageMatches[1].Data, imageMatches[1].Key end
    if #imageMatches > 1 then
        -- ItemModule and Database.Sync.Item can contain duplicate records for
        -- the same icon. Compare their canonical item identity instead of raw
        -- module paths (for example "Vintage.Laser" vs "Laser"). If all
        -- exact icon matches agree on rarity/type, retain the most informative
        -- record so a Retro Bundle counterpart is not mistaken for its
        -- similarly named Vintage original.
        local chosen, chosenScore
        local commonName, commonRarity, commonType, commonChroma
        for _, entry in ipairs(imageMatches) do
            local record = entry.Data
            local recordName = entry.CanonicalName
            local rarity = entry.RarityKey
            local itemType = entry.RecordType
            local chroma = entry.IsChroma
            if commonName and recordName ~= commonName then return nil end
            if commonRarity and rarity ~= "" and rarity ~= commonRarity then return nil end
            if commonType and itemType ~= "" and itemType ~= commonType then return nil end
            if commonChroma ~= nil and chroma ~= commonChroma then return nil end
            commonName = commonName or recordName
            commonRarity = commonRarity or (rarity ~= "" and rarity or nil)
            commonType = commonType or (itemType ~= "" and itemType or nil)
            if commonChroma == nil then commonChroma = chroma end

            local score = (rarity ~= "" and 4 or 0) + (itemType ~= "" and 2 or 0)
                + ((record.Name ~= nil or record.ItemName ~= nil) and 1 or 0)
            if not chosen or score > chosenScore then
                chosen, chosenScore = entry, score
            end
        end
        if chosen then return chosen.Data, chosen.Key end
        return nil
    end
    if #candidates == 1 then return candidates[1].Data, candidates[1].Key end
    if #candidates > 1 then
        local first = candidates[1]
        local firstData, firstKey = first.Data, first.Key
        local rarity = recordRarityKey(firstData, firstKey)
        local itemType = firstData.Type or firstData.ItemType or firstData.WeaponType
        local function variantSignature(record, key)
            local leafKey = tostring(key or ""):match("([^%.]+)$") or tostring(key or "")
            local fields = {leafKey}
            local function addField(value)
                if value ~= nil then table.insert(fields, normalize(tostring(value))) end
            end
            addField(record.Variant)
            addField(record.Year)
            addField(record.Origin)
            addField(record.Event)
            addField(record.ItemId)
            addField(record.ItemID)
            addField(record.Id)
            addField(record.ID)
            addField(record.Image)
            addField(record.ImageId)
            addField(record.ImageID)
            addField(record.Icon)
            addField(record.IconId)
            addField(record.IconImageId)
            addField(record.TextureId)
            return table.concat(fields, "|")
        end
        local firstVariant = variantSignature(firstData, firstKey)
        for _, entry in ipairs(candidates) do
            local record = entry.Data
            local candidateType = record.Type or record.ItemType or record.WeaponType
            if entry.RarityKey ~= rarity or candidateType ~= itemType
                or entry.IsChroma ~= first.IsChroma then return nil end
            -- A name-only match is unsafe when the game catalogue contains
            -- separate same-name variants (for example Scratch 2016/2017).
            -- Require an exact card icon in that case instead of selecting
            -- whichever record happens to be traversed first.
            if variantSignature(record, entry.Key) ~= firstVariant then return nil end
        end
        return firstData, firstKey
    end
    return nil
end

local function hasVisibleTag(itemFrame, tagName)
    local tags = itemFrame and itemFrame:FindFirstChild("Tags", true)
    local tag = tags and tags:FindFirstChild(tagName, true)
    if not tag or not tag:IsA("GuiObject") then return false end
    local current = tag
    while current do
        if current:IsA("GuiObject") and not current.Visible then return false end
        if current == itemFrame then return true end
        current = current.Parent
    end
    return false
end

local function CheckIsChroma(itemFrame, itemName, resolvedRecord, resolvedKey, lookupDone)
    local nameSaysChroma = isChromaName(itemName)
    local abbreviatedChroma = type(itemName) == "string" and string.match(itemName, "^[Cc]%s*%.%s*") ~= nil
    -- The per-card Chroma tag and explicit name take precedence over other
    -- tags; some game cards can expose more than one tag at once.
    if nameSaysChroma or abbreviatedChroma then return true end
    if hasVisibleTag(itemFrame, "Chroma") then return true end
    local record, recordKey = resolvedRecord, resolvedKey
    if not lookupDone and itemName then record, recordKey = findGameItem(itemName, itemFrame) end
    if record then
        local rarity = string.lower(tostring(record.Rarity or record.ItemRarity or record.RarityName or record.Category or ""))
        return rarity == "chroma" or isChromaName(record.Name or record.ItemName) or isChromaName(recordKey)
    end
    -- Some trade-card layouts name the badge "ChromaTag" or expose its text
    -- inside Tags instead of using an instance literally named "Chroma". Use
    -- this deeper fallback only if the indexed game catalogue had no match.
    local tags = itemFrame and itemFrame:FindFirstChild("Tags", true)
    if tags then
        for _, tag in ipairs(tags:GetDescendants()) do
            local tagName = normalize(tag.Name)
            local tagText = (tag:IsA("TextLabel") or tag:IsA("TextButton")) and normalize(tag.Text) or ""
            if (tagName == "chroma" or tagName == "chromatag" or tagText == "chroma") then
                local current = tag
                local visible = true
                while current and current ~= itemFrame do
                    if current:IsA("GuiObject") and not current.Visible then visible = false; break end
                    current = current.Parent
                end
                if visible then return true end
            end
        end
    end
    -- With no matching catalogue record or visible tag, do not guess from the
    -- item's rarity; this avoids turning a normal item into its Chroma variant.
    return false
end

local function getRecordRarity(record, recordKey)
    if type(record) ~= "table" then return nil end
    -- ItemModule versions have used different field names for the class. Only
    -- accept recognized rarity labels; fields such as Type can mean knife/gun.
    for _, field in ipairs({"Rarity", "ItemRarity", "RarityName", "Tier", "Category"}) do
        local value = record[field]
        if type(value) == "string" and value ~= "" then
            local rarity = string.lower(value)
            if rarity == "classic" then return "vintage" end
            if rarity == "unique" or rarity == "ancient" or rarity == "godly" or rarity == "vintage"
                or rarity == "legendary" or rarity == "rare" or rarity == "uncommon" or rarity == "common"
                or rarity == "chroma" then return rarity end
        end
    end
    -- Some catalogues organize variants under a rarity-named key rather than
    -- storing the rarity on the leaf record itself.
    local key = normalize(tostring(recordKey or ""))
    for _, rarity in ipairs({"vintage", "classic", "ancient", "godly", "legendary", "rare", "uncommon", "common", "unique", "chroma"}) do
        if rarity == "chroma" then
            if isChromaName(recordKey) then return "chroma" end
        elseif string.find(key, rarity, 1, true) then
            if rarity == "classic" then return "vintage" end
            return rarity
        end
    end
    return nil
end

local function DetectItemRarity(itemFrame, itemName, resolvedRecord, resolvedKey, lookupDone)
    -- Instance tags and the item's actual container take precedence over a
    -- name-keyed catalogue, which can contain duplicate display names.
    local rarityNames = {"unique", "ancient", "godly", "vintage", "legendary", "rare", "uncommon", "common"}
    local descendants = itemFrame and itemFrame:GetDescendants() or {}
    local visibleRarityTags = {}
    for _, desc in ipairs(descendants) do
        local rarity = normalize(desc.Name)
        if desc:IsA("GuiObject") and (rarity == "classic" or table.find(rarityNames, rarity)) then
            local current, visible, insideTags = desc, true, false
            while current and current ~= itemFrame do
                local currentName = normalize(current.Name)
                if currentName == "tags" then insideTags = true end
                if current:IsA("GuiObject") and not current.Visible then visible = false; break end
                current = current.Parent
            end
            if visible and insideTags then
                visibleRarityTags[rarity == "classic" and "vintage" or rarity] = true
            end
        end
    end
    for _, rarity in ipairs(rarityNames) do
        if visibleRarityTags[rarity] then return rarity end
    end
    -- Some game UI versions expose rarity as an attribute, a named badge, or
    -- a text label outside the Tags folder. Read only exact rarity identifiers;
    -- never infer rarity from the badge color or from the display name.
    if itemFrame then
        local function readRarity(instance)
            for _, attr in ipairs({"Rarity", "ItemRarity", "RarityName"}) do
                local ok, value = pcall(function() return instance:GetAttribute(attr) end)
                if ok and type(value) == "string" then
                    local normalized = normalize(value)
                    if normalized == "classic" then return "vintage" end
                    for _, rarity in ipairs(rarityNames) do
                        if normalized == rarity then return rarity end
                    end
                end
            end
            local normalizedName = normalize(instance.Name)
            if normalizedName == "classic" then return "vintage" end
            for _, rarity in ipairs(rarityNames) do
                if normalizedName == rarity then return rarity end
            end
            if instance:IsA("TextLabel") or instance:IsA("TextButton") then
                local normalizedText = normalize(instance.Text)
                if normalizedText == "classic" then return "vintage" end
                for _, rarity in ipairs(rarityNames) do
                    if normalizedText == rarity then return rarity end
                end
            end
            return nil
        end
        local found = readRarity(itemFrame)
        if found then return found end
        for _, desc in ipairs(descendants) do
            if desc:IsA("GuiObject") and not desc.Visible then
                -- Ignore hidden rarity badges left over from another card state.
            else
                found = readRarity(desc)
                if found then return found end
            end
        end
    end
    local current = itemFrame and itemFrame.Parent
    while current and current ~= CoreGui do
        local name = normalize(current.Name)
        if name == "classic" then return "vintage" end
        for _, rarity in ipairs(rarityNames) do
            if name == rarity or name == rarity .. "s" then return rarity end
        end
        if name == "godlies" then return "godly" end
        current = current.Parent
    end
    local record, recordKey = resolvedRecord, resolvedKey
    if not lookupDone and itemName then record, recordKey = findGameItem(itemName, itemFrame) end
    local recordRarity = getRecordRarity(record, recordKey)
    if recordRarity then return recordRarity end

    -- Exact observed icon IDs are a rarity-only fallback for the two Starry
    -- variants. Prices are never stored here; GetItemData still requires a
    -- matching numeric entry parsed from Supreme Values.
    local starryIconRarities = {
        ["8304772774"] = "rare",
        ["130579251074"] = "uncommon",
    }
    local _, itemImage = getFrameIdentity(itemFrame)
    local iconRarity = starryIconRarities[getImageAssetId(itemImage)]
    if iconRarity then return iconRarity end

    return nil
end

local function getFrameYear(itemFrame)
    if not itemFrame then return nil end
    local function findYear(value)
        return value and string.match(tostring(value), "(20%d%d)") or nil
    end
    local tags = itemFrame:FindFirstChild("Tags", true)
    if tags then
        for _, instance in ipairs(tags:GetDescendants()) do
            local name = normalize(instance.Name)
            if name == "year" or name == "year2" then
                local value
                if instance:IsA("ValueBase") then
                    value = instance.Value
                elseif instance:IsA("TextLabel") or instance:IsA("TextButton") then
                    local current = instance
                    local visible = true
                    while current and current ~= tags do
                        if current:IsA("GuiObject") and not current.Visible then visible = false; break end
                        current = current.Parent
                    end
                    if visible then value = instance.Text end
                end
                local year = findYear(value)
                if year then return year end
            end
        end
    end
    return nil
end

local function GetItemCount(itemFrame, amountLabel)
    local amount = amountLabel or (itemFrame and itemFrame:FindFirstChild("Amount", true))
    if amount and amount:IsA("TextLabel") then
        local digits = string.match(amount.Text, "%d+")
        local count = digits and tonumber(digits)
        if count and count > 0 then return count end
    end
    return 1
end

local function GetItemData(itemName, isChroma, isVintage, itemFrame)
    if type(itemName) ~= "string" or itemName == "" then return nil end
    local gameRecord, gameRecordKey = findGameItem(itemName, itemFrame)
    local recordRarity = getRecordRarity(gameRecord, gameRecordKey)
    local recordName = gameRecord and (gameRecord.Name or gameRecord.ItemName) or nil
    local recordType = gameRecord and normalize(tostring(gameRecord.Type or gameRecord.ItemType or gameRecord.WeaponType or gameRecord.ItemClass or "")) or ""
    if recordType ~= "gun" and recordType ~= "knife" then recordType = "" end
    if recordType == "" then
        local keyType = normalize(tostring(gameRecordKey or ""))
        if string.match(keyType, "gun$") then recordType = "gun"
        elseif string.match(keyType, "knife$") then recordType = "knife" end
    end
    if recordType == "" then recordType = getFrameWeaponType(itemFrame) or "" end
    local recordSaysChroma = recordRarity == "chroma"
        or isChromaName(recordName) or isChromaName(gameRecordKey)
    -- Reuse the resolved game record so one card is matched against the game
    -- catalogue once, not again for chroma and rarity checks.
    isChroma = isChroma or recordSaysChroma or CheckIsChroma(itemFrame, itemName, gameRecord, gameRecordKey, true)
    isVintage = isVintage or recordRarity == "vintage"
    local abbreviatedChromaName = string.match(itemName, "^[Cc]%s*%.%s*(.+)$")
    local baseItemName = abbreviatedChromaName or itemName
    local cleanInput = normalize(itemName)
    if isChroma then
        local baseName = normalize(baseItemName):gsub("chroma", "")
        cleanInput = "chroma" .. baseName
    end

    local detectedRarity = DetectItemRarity(itemFrame, itemName, gameRecord, gameRecordKey, true)
    if not detectedRarity then detectedRarity = recordRarity end
    if isChroma then detectedRarity = "chroma" end
    if isVintage then detectedRarity = "vintage" end
    local targetYear = getFrameYear(itemFrame)
    if not targetYear and gameRecord then
        for _, field in ipairs({"Year", "Origin", "Event", "Variant"}) do
            local value = gameRecord[field]
            targetYear = value and string.match(tostring(value), "(20%d%d)") or nil
            if targetYear then break end
        end
    end
    if not targetYear then
        targetYear = string.match(tostring(gameRecordKey or ""), "(20%d%d)")
    end
    local function categoryOf(item)
        return string.lower(tostring(item and item.Category or ""))
    end
    local function isValidVariant(item)
        local category = categoryOf(item)
        if isChroma then
            if category ~= "chroma" then return false end
        else
            -- Chroma and regular versions share a display name on Supreme Values.
            -- Never let the earlier-loaded Chroma page supply a regular item's value.
            if category == "chroma" then return false end
            if detectedRarity ~= nil and category ~= detectedRarity then return false end
        end
        if targetYear then
            local itemYear = tostring(item.Year or "")
            if itemYear == "" then itemYear = string.match(tostring(item.Origin or ""), "(20%d%d)") or "" end
            if itemYear ~= targetYear then return false end
        end
        return true
    end

    -- Supreme labels some weapons with a type suffix, e.g. "Starry (Gun)".
    -- The game card may only say "Starry", so use the matched game record's
    -- actual type to query the corresponding Supreme entry.
    local lookupNames = {}
    local function addLookupName(value)
        local key = normalize(tostring(value or ""))
        if key ~= "" then
            for _, existing in ipairs(lookupNames) do
                if existing == key then return end
            end
            table.insert(lookupNames, key)
        end
    end
    -- Supreme separates the same in-game display name into color variants.
    -- Only use that title when the exact matched game record/key identifies
    -- the color; never default every Scratch card to Red Scratch. A generic
    -- "Scratch" database key is deliberately not used because it could be
    -- either Supreme variant and would create a silent wrong-price match.
    local resolvedScratchName
    if cleanInput == "scratch" then
        local recordIdentityParts = {}
        for _, value in pairs({
            gameRecordKey, recordName,
            targetYear,
            gameRecord and gameRecord.Variant,
            gameRecord and gameRecord.Year,
            gameRecord and gameRecord.Origin,
            gameRecord and gameRecord.Event
        }) do
            if value ~= nil then table.insert(recordIdentityParts, tostring(value)) end
        end
        local recordIdentity = normalize(table.concat(recordIdentityParts, " "))
        if string.find(recordIdentity, "blue", 1, true) or string.find(recordIdentity, "2017", 1, true) then
            resolvedScratchName = "Blue Scratch"
            addLookupName(resolvedScratchName)
        elseif string.find(recordIdentity, "red", 1, true) or string.find(recordIdentity, "2016", 1, true) then
            resolvedScratchName = "Red Scratch"
            addLookupName(resolvedScratchName)
        end
    else
        addLookupName(cleanInput)
        addLookupName(baseItemName)
        addLookupName(recordName)
    end
    if recordType ~= "" then
        if cleanInput ~= "scratch" then
            addLookupName((recordName or baseItemName) .. " (" .. recordType .. ")")
            addLookupName(baseItemName .. " (" .. recordType .. ")")
        elseif resolvedScratchName then
            addLookupName(resolvedScratchName .. " (" .. recordType .. ")")
        end
    end
    if isChroma then
        local chromaBase = normalize(baseItemName):gsub("^chroma", ""):gsub("chroma$", "")
        addLookupName("Chroma " .. chromaBase)
        addLookupName("C. " .. chromaBase)
        if recordType ~= "" then addLookupName("Chroma " .. chromaBase .. " (" .. recordType .. ")") end
    end

    local variants
    for _, lookupName in ipairs(lookupNames) do
        local candidateList = Database[lookupName]
        if candidateList then
            local matching = {}
            if candidateList.Value then
                if isValidVariant(candidateList) then
                    table.insert(matching, candidateList)
                end
            else
                for _, entry in ipairs(candidateList) do
                    if isValidVariant(entry) then
                        table.insert(matching, entry)
                    end
                end
            end
            if #matching > 0 then
                variants = matching
                break
            end
        end
    end

    -- Keep Corrupt's alternate Supreme label handling, but never invent a
    -- value: the record must still have come from Supreme's Unique page.
    if not variants and cleanInput == "corrupt" then
        local list = Database.corrupt
        if list then
            variants = {}
            for _, entry in ipairs(list) do
                if categoryOf(entry) == "unique" and isValidVariant(entry) then
                    table.insert(variants, entry)
                end
            end
        end
    end

    if variants and #variants > 0 then
        local first = variants[1]
        if not tonumber(first.Value) or tonumber(first.Value) <= 0 then
            return {Value = 0, Stability = "N/A", Category = detectedRarity or (isChroma and "chroma" or categoryOf(first))}
        end
        for i = 2, #variants do
            if not tonumber(variants[i].Value) or tonumber(variants[i].Value) <= 0
                or tonumber(variants[i].Value) ~= tonumber(first.Value) or variants[i].Stability ~= first.Stability then
                return {Value = 0, Stability = "N/A", Category = detectedRarity or (isChroma and "chroma" or "unique")}
            end
        end
        return first
    end
    return {Value = 0, Stability = "N/A", Category = detectedRarity or (isChroma and "chroma" or "unique")}
end

local function GetItemCardButton(itemFrame)
    if not itemFrame then return nil end
    if itemFrame:IsA("GuiButton") and itemFrame.Visible and itemFrame.Active then return itemFrame end
    -- In the phone trade inventory the actual add-to-trade control is
    -- NewItem.Container.ActionButton. Prefer this exact control when present.
    local cardContainer = itemFrame:FindFirstChild("Container")
    local actionButton = cardContainer and cardContainer:FindFirstChild("ActionButton", true)
    if actionButton and actionButton:IsA("GuiButton") and actionButton.Visible and actionButton.Active then
        return actionButton
    end
    for _, desc in ipairs(itemFrame:GetDescendants()) do
        if desc:IsA("GuiButton") and desc.Visible and desc.Active then return desc end
    end
    -- Some game builds put the click button beside the title/card frame.
    -- Use it only when the card center is actually inside that button.
    local frameParent = itemFrame.Parent
    if frameParent then
        local center = itemFrame.AbsolutePosition + itemFrame.AbsoluteSize / 2
        for _, sibling in ipairs(frameParent:GetChildren()) do
            if sibling:IsA("GuiButton") and sibling.Visible and sibling.Active then
                local pos, size = sibling.AbsolutePosition, sibling.AbsoluteSize
                if center.X >= pos.X and center.Y >= pos.Y
                    and center.X < pos.X + size.X and center.Y < pos.Y + size.Y then
                    return sibling
                end
            end
        end
    end
    return nil
end

local function GetVirtualInputManager()
    local ok, input = pcall(function()
        return game:GetService("VirtualInputManager")
    end)
    if ok and input then
        local methodOk, method = pcall(function() return input.SendMouseButtonEvent end)
        if methodOk and type(method) == "function" then return input end
    end
    return nil
end

local function CanClickItem(itemFrame)
    if not itemFrame or not itemFrame.Parent then return false end
    if GetItemCardButton(itemFrame) and type(ExecutorEnv.firesignal) == "function" then return true end
    return GetVirtualInputManager() ~= nil
end

local function ClickItem(itemFrame, forceMouse)
    if not itemFrame or not itemFrame.Parent then return false end
    local button = GetItemCardButton(itemFrame)

    -- Try the same signal used by the game's inventory buttons, then allow
    -- the caller to retry with a real screen click if the offer does not change.
    if not forceMouse and button and type(ExecutorEnv.firesignal) == "function" then
        local ok = pcall(function() ExecutorEnv.firesignal(button.MouseButton1Click) end)
        if not ok then
            ok = pcall(function() ExecutorEnv.firesignal(button.Activated) end)
        end
        if ok then return true end
    end

    local input = GetVirtualInputManager()
    local target = button or itemFrame
    if not input or not target.Visible then return false end
    local position = target.AbsolutePosition + target.AbsoluteSize / 2
    if target.AbsoluteSize.X <= 0 or target.AbsoluteSize.Y <= 0 then return false end
    local camera = workspace.CurrentCamera
    local viewport = camera and camera.ViewportSize
    if viewport and (position.X < 0 or position.Y < 0 or position.X >= viewport.X or position.Y >= viewport.Y) then
        return false
    end

    local downOk = pcall(function()
        input:SendMouseButtonEvent(position.X, position.Y, 0, true, game, 0)
    end)
    if not downOk then
        downOk = pcall(function()
            input:SendMouseButtonEvent(position.X, position.Y, Enum.UserInputType.MouseButton1, true, game, 0)
        end)
        if downOk then
            task.wait(0.04)
            pcall(function()
                input:SendMouseButtonEvent(position.X, position.Y, Enum.UserInputType.MouseButton1, false, game, 0)
            end)
            return true
        end
        return false
    end
    task.wait(0.04)
    local upOk = pcall(function()
        input:SendMouseButtonEvent(position.X, position.Y, 0, false, game, 0)
    end)
    return upOk
end

local function GetActiveTradeGui(playerGui, allowDisabled)
    if not playerGui then return nil end
    -- The layout setting maps phone to TradeGUI_Phone and PC/tablet to TradeGUI.
    -- Do not fall back to the other GUI: both can exist in PlayerGui, and
    -- choosing the wrong one makes the inventory scan look empty.
    if TargetGuiName ~= "TradeGUI" and TargetGuiName ~= "TradeGUI_Phone" then return nil end
    local gui = playerGui:FindFirstChild(TargetGuiName)
    if not gui then return nil end
    if not allowDisabled and gui:IsA("ScreenGui") and not gui.Enabled then return nil end
    return gui
end

local function SelectWeaponsTab(itemsPanel)
    local tabs = itemsPanel and (itemsPanel:FindFirstChild("Tabs") or itemsPanel:FindFirstChild("Tabs", true))
    if not tabs then return false end
    for _, descendant in ipairs(tabs:GetDescendants()) do
        local text = descendant:IsA("TextLabel") or descendant:IsA("TextButton") or descendant:IsA("TextBox")
        local name = string.lower(tostring(descendant.Name or ""))
        local value = text and string.lower(tostring(descendant.Text or "")) or ""
        if name:find("weapon", 1, true) or value:find("weapon", 1, true)
            or value:find("оруж", 1, true) then
            local current = descendant
            while current and current ~= tabs do
                if current:IsA("GuiButton") then
                    if ClickItem(current) then
                        task.wait(0.2)
                        return true
                    end
                    break
                end
                current = current.Parent
            end
        end
    end
    return false
end

local function GetInventoryItems(progressLabel, includeHidden)
    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    if not playerGui then return {}, nil end

    local tradeGui = GetActiveTradeGui(playerGui, includeHidden)
    local container = tradeGui and tradeGui:FindFirstChild("Container")
    local itemsPanel = container and container:FindFirstChild("Items")
    if not itemsPanel then return {}, nil end

    -- The phone inventory keeps weapon cards at:
    -- Items.Main.Weapons.Items.Container.Current.Container
    -- These cards are all named "NewItem" (not NewItem1/NewItem2), so scan
    -- that card container directly when the hierarchy is available.
    local inventoryRoot = itemsPanel
    local path = {"Main", "Weapons", "Items", "Container", "Current", "Container"}
    local current = itemsPanel
    local foundPhoneCards = true
    for _, childName in ipairs(path) do
        current = current and current:FindFirstChild(childName)
        if not current then
            foundPhoneCards = false
            break
        end
    end
    if foundPhoneCards then inventoryRoot = current end

    local items = {}
    local pendingCards = {}
    local recognizedCards = 0
    local unpricedNames = {}
    local seenCards = {}
    local itemLabelCache = setmetatable({}, {__mode = "k"})
    local cardIconCache = setmetatable({}, {__mode = "k"})

    local function isActuallyVisible(card)
        if not card:IsDescendantOf(inventoryRoot) then return false end
        if includeHidden then return true end
        local scan = card
        while scan do
            if scan:IsA("GuiObject") and not scan.Visible then return false end
            if scan == inventoryRoot then break end
            scan = scan.Parent
        end
        return scan == inventoryRoot
    end

    local function itemLabel(card)
        local cached = itemLabelCache[card]
        if cached ~= nil then return cached or nil end
        local nameContainer = card:FindFirstChild("ItemName") or card:FindFirstChild("NameLabel") or card:FindFirstChild("Label")
        if not nameContainer then
            nameContainer = card:FindFirstChild("ItemName", true) or card:FindFirstChild("NameLabel", true)
        end
        if not nameContainer then itemLabelCache[card] = false; return nil end
        if nameContainer:IsA("TextLabel") then itemLabelCache[card] = nameContainer; return nameContainer end
        local label = nameContainer:FindFirstChild("Label", true) or nameContainer:FindFirstChild("TextLabel", true)
        if label and label:IsA("TextLabel") then itemLabelCache[card] = label; return label end
        itemLabelCache[card] = false
        return nil
    end

    local function hasCardIcon(card)
        local cached = cardIconCache[card]
        if cached ~= nil then return cached or nil end
        local cardContainer = card:FindFirstChild("Container")
        local icon = cardContainer and cardContainer:FindFirstChild("Icon", true)
        if not icon then icon = card:FindFirstChild("Icon", true) end
        local valid = icon and (icon:IsA("ImageLabel") or icon:IsA("ImageButton"))
        cardIconCache[card] = valid or false
        return valid
    end

    local function isItemCard(card)
        -- Desktop revisions may use numbered cards, while the phone UI uses
        -- repeated "NewItem" names. Identify the latter by its name, icon,
        -- and item-name structure rather than relying on a unique Instance name.
        if string.match(card.Name, "^NewItem%d+$") then return true end
        local directName = card:FindFirstChild("ItemName") or card:FindFirstChild("NameLabel")
        local directContainer = card:FindFirstChild("Container")
        return directName ~= nil and directContainer ~= nil and hasCardIcon(card)
    end

    local function considerCard(card, labelOverride)
        if not card:IsA("GuiObject") then return false end
        if not labelOverride and not isItemCard(card) then return false end
        if not isActuallyVisible(card) or not hasCardIcon(card) then return false end
        local label = labelOverride or itemLabel(card)
        local itemName = label and label.Text and label.Text:gsub("^%s+", ""):gsub("%s+$", "")
        if not label then return false end
        if seenCards[card] then return true end
        if not isDatabaseLoaded then
            -- Snapshot card references and counts immediately at startup. The
            -- value catalogue may still be loading; price these same cards
            -- later without walking the inventory again.
            seenCards[card] = true
            pendingCards[#pendingCards + 1] = {
                Frame = card,
                Label = label,
                Name = itemName,
                Count = GetItemCount(card)
            }
            recognizedCards = recognizedCards + 1
            return true
        end
        if not itemName or itemName == "" or itemName == "Item" or itemName == "Loading" then return false end

        local data = GetItemData(itemName, false, false, card)
        if data and tonumber(data.Value) and data.Value > 0 then
            seenCards[card] = true
            recognizedCards = recognizedCards + 1
            local itemId, itemImage = getFrameIdentity(card)
            local count = GetItemCount(card)
            table.insert(items, {
                Frame = card,
                Name = itemName,
                ItemId = itemId,
                ImageId = getImageAssetId(itemImage),
                Category = data.Category or "",
                Count = count,
                UnitValue = tonumber(data.Value),
                Value = tonumber(data.Value) * count,
                Stability = data.Stability or "Stable"
            })
            return true
        end
        local labelName = string.lower(tostring(label.Name or ""))
        local isNameLabel = itemLabel(card) == label or labelName == "label" or labelName:find("name", 1, true)
        if isNameLabel then
            recognizedCards = recognizedCards + 1
            if itemLabel(card) == label then seenCards[card] = true end
        end
        if isNameLabel and #unpricedNames < 4 then
            local alreadyListed = false
            for _, previousName in ipairs(unpricedNames) do
                if previousName == itemName then alreadyListed = true; break end
            end
            if not alreadyListed then unpricedNames[#unpricedNames + 1] = itemName end
        end
        return false
    end

    local scanSteps = 0
    if foundPhoneCards then
        -- Every direct child here is one phone NewItem card. Avoid recursively
        -- walking each card's labels, tags, and layout objects.
        local cards = inventoryRoot:GetChildren()
        for index, card in ipairs(cards) do
            if card:IsA("GuiObject") then considerCard(card) end
            scanSteps = scanSteps + 1
            if progressLabel and index % 12 == 0 then
                setTradeStatus(string.format("%s %d / %d", progressLabel, index, #cards))
            end
            if scanSteps % 4 == 0 then task.wait() end
        end
    else
        -- One flat hierarchy snapshot replaces a recursive GetChildren walk
        -- followed by a second GetDescendants walk. The latter list also feeds
        -- the legacy geometry fallback below.
        local descendants = inventoryRoot:GetDescendants()
        for index, descendant in ipairs(descendants) do
            if descendant:IsA("GuiObject") then
                considerCard(descendant)
            end
            if index % 48 == 0 then task.wait() end
        end

        -- Geometry-based grouping is only needed for legacy card layouts.
        for index, descendant in ipairs(descendants) do
            if descendant:IsA("TextLabel") and descendant.Text ~= "" then
                local card, scan, alreadyHandled = nil, descendant.Parent, false
                while scan and scan ~= inventoryRoot do
                    if seenCards[scan] then
                        alreadyHandled = true
                        break
                    end
                    if scan:IsA("GuiObject") then
                        local size = scan.AbsoluteSize
                        if size.X > 220 or size.Y > 220 then break end
                        if size.X >= 25 and size.Y >= 25 and hasCardIcon(scan) then
                            card = scan
                        end
                    end
                    scan = scan.Parent
                end
                if not alreadyHandled and card and isActuallyVisible(card) then
                    -- Only keep the card when this label resolves to an actual
                    -- valued item. Rarity/tag text in the same tile is ignored.
                    local wasSeen = seenCards[card]
                    if not wasSeen then
                        local directName = itemLabel(card)
                        if directName then
                            considerCard(card, directName)
                        else
                            considerCard(card, descendant)
                        end
                    end
                end
            end
            if index % 80 == 0 then task.wait() end
        end
    end
    return items, inventoryRoot, recognizedCards, unpricedNames, foundPhoneCards, pendingCards
end

-- Warm the inventory cache in the background as soon as the value database
-- and selected trade GUI are available. The generator can then reuse these
-- item records instead of walking every card while the trade is in progress.
local InventoryCache = {
    Scanned = false,
    DatabaseReady = false,
    Ready = false,
    Dirty = true,
    TargetName = nil,
    Root = nil,
    Items = {},
    Groups = {},
    Pending = {},
    CardQueue = {},
    CardQueueRunning = false,
    CardRetryCount = setmetatable({}, {__mode = "k"}),
    Recognized = 0,
    Unpriced = {},
    PhoneCards = false,
    Paused = false,
    DirtyDuringGeneration = false,
    ExpectedFrames = {},
    ConsumedFrames = {},
    Connections = {}
}

local function disconnectInventoryCache()
    for _, connection in ipairs(InventoryCache.Connections) do
        pcall(function() connection:Disconnect() end)
    end
    table.clear(InventoryCache.Connections)
end

local function summarizePricedItems(items)
    local groups = {}
    for _, item in ipairs(items or {}) do
        -- Include identity and rarity so same-named variants remain separate.
        local key = table.concat({
            normalize(item.Name), tostring(item.ItemId or ""), tostring(item.ImageId or ""),
            normalize(item.Category), tostring(item.UnitValue or 0)
        }, "|")
        local group = groups[key]
        if not group then
            group = {
                Name = item.Name,
                Category = item.Category,
                UnitValue = item.UnitValue,
                Quantity = 0,
                Stacks = 0,
                TotalValue = 0
            }
            groups[key] = group
        end
        group.Quantity = group.Quantity + (item.Count or 1)
        group.Stacks = group.Stacks + 1
        group.TotalValue = group.TotalValue + (item.Value or 0)
    end
    return groups
end

local function readPricedInventoryCard(card, root)
    if not card or not card:IsA("GuiObject") or not root or not card:IsDescendantOf(root) then
        return nil, nil
    end
    local container = card:FindFirstChild("Container")
    local icon = container and container:FindFirstChild("Icon", true)
    if not icon then icon = card:FindFirstChild("Icon", true) end
    if not icon or not (icon:IsA("ImageLabel") or icon:IsA("ImageButton")) then return nil, nil end

    local nameNode = card:FindFirstChild("ItemName") or card:FindFirstChild("NameLabel")
        or card:FindFirstChild("Label") or card:FindFirstChild("ItemName", true)
        or card:FindFirstChild("NameLabel", true)
    local label = nameNode and (nameNode:IsA("TextLabel") and nameNode
        or nameNode:FindFirstChild("Label", true)
        or nameNode:FindFirstChild("TextLabel", true))
    if not label or not label:IsA("TextLabel") then return nil, nil, true end

    local itemName = tostring(label.Text or ""):gsub("^%s+", ""):gsub("%s+$", "")
    if itemName == "" or itemName == "Item" or itemName == "Loading" then return nil, nil, true end
    local data = GetItemData(itemName, false, false, card)
    if not data or not tonumber(data.Value) or data.Value <= 0 then
        return nil, itemName
    end

    local itemId, itemImage = getFrameIdentity(card)
    local count = GetItemCount(card)
    return {
        Frame = card,
        Name = itemName,
        ItemId = itemId,
        ImageId = getImageAssetId(itemImage),
        Category = data.Category or "",
        Count = count,
        UnitValue = tonumber(data.Value),
        Value = tonumber(data.Value) * count,
        Stability = data.Stability or "Stable"
    }, itemName
end

local function queueInventoryCardMutation(card)
    if not card then return end
    if InventoryCache.Paused then
        InventoryCache.DirtyDuringGeneration = true
        InventoryCache.Dirty = true
        return
    end
    InventoryCache.CardQueue[card] = true
    if InventoryCache.CardQueueRunning then return end
    InventoryCache.CardQueueRunning = true
    task.spawn(function()
        -- Coalesce the many DescendantAdded events emitted while one card is
        -- constructed, then update only those cards instead of rescanning the
        -- entire inventory tree when TradeGUI opens.
        task.wait(0.1)
        while next(InventoryCache.CardQueue) do
            while InventoryCache.Paused do task.wait(0.1) end
            local changedCards = InventoryCache.CardQueue
            InventoryCache.CardQueue = {}
            local retained = {}
            for _, item in ipairs(InventoryCache.Items) do
                if not changedCards[item.Frame] then
                    retained[#retained + 1] = item
                end
            end
            InventoryCache.Items = retained

            local processed = 0
            for changedCard in pairs(changedCards) do
                if changedCard.Parent and InventoryCache.Root and changedCard:IsDescendantOf(InventoryCache.Root) then
                    local entry, itemName, waitingForCard = readPricedInventoryCard(changedCard, InventoryCache.Root)
                    if entry then
                        InventoryCache.CardRetryCount[changedCard] = nil
                        InventoryCache.Items[#InventoryCache.Items + 1] = entry
                        InventoryCache.Recognized = InventoryCache.Recognized + 1
                    elseif itemName then
                        InventoryCache.CardRetryCount[changedCard] = nil
                        InventoryCache.Recognized = InventoryCache.Recognized + 1
                        if #InventoryCache.Unpriced < 4 then
                            local alreadyListed = false
                            for _, previousName in ipairs(InventoryCache.Unpriced) do
                                if previousName == itemName then alreadyListed = true; break end
                            end
                            if not alreadyListed then InventoryCache.Unpriced[#InventoryCache.Unpriced + 1] = itemName end
                        end
                    elseif waitingForCard then
                        local retries = (InventoryCache.CardRetryCount[changedCard] or 0) + 1
                        InventoryCache.CardRetryCount[changedCard] = retries
                        if retries <= 5 then
                            task.delay(0.2, function()
                                if changedCard.Parent and InventoryCache.Root and changedCard:IsDescendantOf(InventoryCache.Root) then
                                    queueInventoryCardMutation(changedCard)
                                else
                                    InventoryCache.CardRetryCount[changedCard] = nil
                                end
                            end)
                        else
                            InventoryCache.CardRetryCount[changedCard] = nil
                        end
                    end
                end
                processed = processed + 1
                if processed % 8 == 0 then task.wait() end
            end
            table.sort(InventoryCache.Items, function(a, b) return a.Value > b.Value end)
            InventoryCache.Groups = summarizePricedItems(InventoryCache.Items)
            InventoryCache.Ready = #InventoryCache.Items > 0
            InventoryCache.DatabaseReady = isDatabaseLoaded
            InventoryCache.Scanned = true
            InventoryCache.Dirty = false
        end
        InventoryCache.CardQueueRunning = false
    end)
end

local function pricePendingInventoryCache()
    local items, unpricedNames, recognized = {}, {}, 0
    local root = InventoryCache.Root
    for index, pending in ipairs(InventoryCache.Pending) do
        local card, label = pending.Frame, pending.Label
        if card and root and card.Parent and card:IsDescendantOf(root) then
            if not label or not label.Parent or not label:IsDescendantOf(card) then
                label = card:FindFirstChild("ItemName", true)
                    or card:FindFirstChild("NameLabel", true)
                    or card:FindFirstChild("Label", true)
                if label and not label:IsA("TextLabel") then
                    label = label:FindFirstChild("Label", true)
                        or label:FindFirstChild("TextLabel", true)
                end
            end
        end
        if card and label and root and card.Parent and card:IsDescendantOf(root)
            and label.Parent and label:IsDescendantOf(card) and label:IsA("TextLabel") then
            local itemName = tostring(label.Text or ""):gsub("^%s+", ""):gsub("%s+$", "")
            if itemName ~= "" and itemName ~= "Item" and itemName ~= "Loading" then
                recognized = recognized + 1
                local data = GetItemData(itemName, false, false, card)
                if data and tonumber(data.Value) and data.Value > 0 then
                    local itemId, itemImage = getFrameIdentity(card)
                    local count = GetItemCount(card)
                    items[#items + 1] = {
                        Frame = card,
                        Name = itemName,
                        ItemId = itemId,
                        ImageId = getImageAssetId(itemImage),
                        Category = data.Category or "",
                        Count = count,
                        UnitValue = tonumber(data.Value),
                        Value = tonumber(data.Value) * count,
                        Stability = data.Stability or "Stable"
                    }
                elseif #unpricedNames < 4 then
                    local duplicate = false
                    for _, previousName in ipairs(unpricedNames) do
                        if previousName == itemName then duplicate = true; break end
                    end
                    if not duplicate then unpricedNames[#unpricedNames + 1] = itemName end
                end
            end
        end
        if index % 32 == 0 then task.wait() end
    end
    table.sort(items, function(a, b) return a.Value > b.Value end)
    InventoryCache.Items = items
    InventoryCache.Groups = summarizePricedItems(items)
    InventoryCache.Unpriced = unpricedNames
    InventoryCache.Recognized = recognized
    InventoryCache.Pending = {}
    InventoryCache.Ready = #items > 0
    InventoryCache.Scanned = true
    InventoryCache.DatabaseReady = true
    InventoryCache.Dirty = false
    return InventoryCache.Ready
end

local function rebuildInventoryCache()
    if isDatabaseLoaded and InventoryCache.Scanned and not InventoryCache.DatabaseReady
        and not InventoryCache.Dirty and InventoryCache.TargetName == TargetGuiName
        and InventoryCache.Root and InventoryCache.Root.Parent then
        return pricePendingInventoryCache()
    end

    local ok, items, root, recognized, unpriced, phoneCards, pendingCards = pcall(GetInventoryItems, nil, true)
    if not ok or not root then return false end
    -- Sort the priced cards once during the background scan. The generator's
    -- hot path can then take a snapshot without sorting the inventory again.
    table.sort(items or {}, function(a, b) return a.Value > b.Value end)

    if InventoryCache.Root ~= root or InventoryCache.TargetName ~= TargetGuiName then
        disconnectInventoryCache()
        InventoryCache.Root = root
        InventoryCache.TargetName = TargetGuiName
        local function invalidate(instance)
            if InventoryCache.Paused then
                if not InventoryCache.ExpectedFrames[instance] then
                    InventoryCache.DirtyDuringGeneration = true
                    InventoryCache.Dirty = true
                end
            elseif InventoryCache.DatabaseReady then
                queueInventoryCardMutation(instance)
            else
                InventoryCache.Dirty = true
            end
        end
        local function owningItemCard(instance)
            local scan = instance
            while scan and scan ~= root do
                if scan.Name == "NewItem" or string.match(scan.Name, "^NewItem%d+$") then
                    return scan
                end
                scan = scan.Parent
            end
            return nil
        end
        table.insert(InventoryCache.Connections, root.DescendantAdded:Connect(function(instance)
            local card = owningItemCard(instance)
            if card then invalidate(card) end
        end))
        table.insert(InventoryCache.Connections, root.DescendantRemoving:Connect(function(instance)
            local card = owningItemCard(instance)
            if card then invalidate(card) end
        end))
    end

    if not isDatabaseLoaded then
        InventoryCache.Items = {}
        InventoryCache.Groups = {}
        InventoryCache.Pending = pendingCards or {}
        InventoryCache.Recognized = recognized or 0
        InventoryCache.Unpriced = {}
        InventoryCache.PhoneCards = phoneCards == true
        InventoryCache.Ready = false
        InventoryCache.Scanned = true
        InventoryCache.DatabaseReady = false
        InventoryCache.Dirty = false
        return false
    end

    InventoryCache.Items = items or {}
    InventoryCache.Groups = summarizePricedItems(InventoryCache.Items)
    InventoryCache.Pending = {}
    InventoryCache.Recognized = recognized or 0
    InventoryCache.Unpriced = unpriced or {}
    InventoryCache.PhoneCards = phoneCards == true
    InventoryCache.Ready = #InventoryCache.Items > 0
    InventoryCache.Scanned = true
    InventoryCache.DatabaseReady = isDatabaseLoaded
    InventoryCache.Dirty = false
    return InventoryCache.Ready
end

local function getInventorySnapshot()
    if InventoryCache.Scanned and InventoryCache.DatabaseReady
        and InventoryCache.TargetName == TargetGuiName and InventoryCache.Root
        and InventoryCache.Root.Parent and not InventoryCache.CardQueueRunning
        and not next(InventoryCache.CardQueue) then
        local snapshot = table.clone(InventoryCache.Items)
        return snapshot, InventoryCache.Root, InventoryCache.Recognized,
            InventoryCache.Unpriced, InventoryCache.PhoneCards
    end
    return nil
end

task.spawn(function()
    while hubAlive do
        local gui = Hub.WindowGui
        if not gui or not gui.Parent then
            -- buildWindow() runs later in [S9]; wait for it instead of
            -- breaking out of the loop on the very first iteration (which
            -- permanently killed the warm cache in earlier builds).
            if not hubAlive then break end
            task.wait(0.5)
        elseif InventoryCache.Paused then
            task.wait(0.25)
        elseif InventoryCache.TargetName ~= TargetGuiName or not InventoryCache.Scanned
            or InventoryCache.Dirty
            or (isDatabaseLoaded and (not InventoryCache.DatabaseReady
                or not InventoryCache.Root or not InventoryCache.Root.Parent)) then
            rebuildInventoryCache()
            task.wait(InventoryCache.Ready and 2 or 0.5)
        else
            task.wait(2)
        end
    end
    disconnectInventoryCache()
end)

-- ==========================================================================
-- [S6] OFFER GENERATOR + TRADE VERDICT + INVENTORY ANALYSIS
-- ==========================================================================
local function AutoSelectOfferNow(mode)
    if not isDatabaseLoaded then
        setTradeStatus("The value list is still loading.")
        return
    end
    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    local tradeGui = GetActiveTradeGui(playerGui)
    local tradeContainer = tradeGui and tradeGui:FindFirstChild("Container")
    local tradeMain = tradeContainer and (tradeContainer:FindFirstChild("Trade") or tradeContainer:FindFirstChild("Main") or tradeContainer)
    local yourOffer = tradeMain and tradeMain:FindFirstChild("YourOffer")
    yourOffer = yourOffer and (yourOffer:FindFirstChild("Container") or yourOffer)
    local theirOffer = tradeMain and tradeMain:FindFirstChild("TheirOffer")
    theirOffer = theirOffer and (theirOffer:FindFirstChild("Container") or theirOffer)
    if not yourOffer or not theirOffer then
        setTradeStatus("Open a trade to build an offer.")
        return
    end

    -- Offer cards and their rarity tags are stable while a trade is open.
    -- Cache each card's unit value so the generator's short polling loop only
    -- reads the current item count instead of resolving catalogue data again.
    local offerValueCache = setmetatable({}, {__mode = "k"})
    local function readOfferState(offer)
        local total, slots = 0, 0
        for _, frame in ipairs(offer:GetChildren()) do
            if not frame:IsA("GuiObject") or not frame.Visible then
                -- hidden slot
            else
                local cached = offerValueCache[frame]
                local label = cached and cached.Label
                if not label or not label.Parent or not label:IsDescendantOf(frame) then
                    local nameNode = frame:FindFirstChild("ItemName", true)
                        or frame:FindFirstChild("NameLabel", true)
                        or frame:FindFirstChild("Label", true)
                    label = nameNode and (nameNode:IsA("TextLabel") and nameNode
                        or nameNode:FindFirstChild("Label", true)
                        or nameNode:FindFirstChild("TextLabel", true))
                end
                if label and label:IsA("TextLabel") and label.Text ~= "" and label.Text ~= "Item" then
                    slots = slots + 1
                    local icon = cached and cached.Icon
                    if not icon or not icon.Parent or not icon:IsDescendantOf(frame) then
                        local cardContainer = frame:FindFirstChild("Container", true)
                        icon = cardContainer and cardContainer:FindFirstChild("Icon", true)
                        if not icon then icon = frame:FindFirstChild("Icon", true) end
                    end
                    local image = icon and (icon:IsA("ImageLabel") or icon:IsA("ImageButton")) and icon.Image or nil
                    if not cached or cached.Name ~= label.Text or cached.Image ~= image then
                        local data = GetItemData(label.Text, false, false, frame)
                        cached = {
                            Name = label.Text,
                            Label = label,
                            Icon = icon,
                            Image = image,
                            Amount = frame:FindFirstChild("Amount", true),
                            Value = data and tonumber(data.Value) or 0
                        }
                    else
                        cached.Label = label
                        cached.Icon = icon
                        if not cached.Amount or not cached.Amount:IsDescendantOf(frame) then
                            cached.Amount = frame:FindFirstChild("Amount", true)
                        end
                    end
                    offerValueCache[frame] = cached
                    if cached.Value > 0 then
                        total = total + cached.Value * GetItemCount(frame, cached.Amount)
                    end
                end
            end
        end
        return total, slots
    end

    local theirValue = readOfferState(theirOffer)
    if theirValue <= 0 then
        setTradeStatus("The other offer has no valued items yet.")
        return
    end

    local factor = mode == "WIN" and 0.75 or (mode == "LOSE" and 1.25 or 1.0)
    local target = theirValue * factor
    local currentOfferValue = readOfferState(yourOffer)
    if currentOfferValue >= target - 0.5 then
        setTradeStatus(string.format("Your offer already meets the target (%d / %d).", currentOfferValue, math.floor(target)))
        return
    end
    setTradeStatus("Using pre-scanned inventory...")
    local candidates, inventoryRoot, recognizedCards, unpricedNames = getInventorySnapshot()
    if not candidates then
        -- Let the background startup scan finish. Do not start a costly scan
        -- on the trade button's thread while the trade is already active.
        for _ = 1, 30 do
            task.wait(0.1)
            candidates, inventoryRoot, recognizedCards, unpricedNames = getInventorySnapshot()
            if candidates then break end
        end
    end
    if not candidates then
        setTradeStatus("Inventory scan is still preparing. Wait a moment, then try again.")
        return
    end
    if not inventoryRoot then
        setTradeStatus("Could not find the inventory panel in the selected trade interface.")
        return
    end
    if #candidates == 0 then
        if recognizedCards and recognizedCards > 0 then
            setTradeStatus(string.format("Found %d item cards, but no values matched: %s. Refresh the value list or send these names.", recognizedCards, table.concat(unpricedNames or {}, ", ")))
        else
            setTradeStatus("No inventory cards were recognized in the selected trade panel.")
        end
        return
    end

    setTradeStatus("Finding a close match and watching for offer changes...")
    -- Plan within the four-slot limit. Keep the remaining bundle while the live
    -- totals still match it; only recompute when either offer changes.
    local maxItems = 4
    local placed, failedName = 0, nil
    local usedFrames, selectedItems = {}, {}
    local actualTarget = target
    local windowWasEnabled = Hub.WindowGui.Enabled
    Hub.WindowGui.Enabled = false
    task.wait(0.08)

    local function findBestBundle(remaining, availableSlots)
        local eligible = {}
        for _, item in ipairs(candidates) do
            local frame = item.Frame
            if item.Value > 0 and item.Value <= remaining and frame and frame.Parent
                and frame:IsDescendantOf(inventoryRoot) and frame.Visible and not usedFrames[frame] then
                eligible[#eligible + 1] = item
            end
        end
        if #eligible == 0 or availableSlots <= 0 then return nil end

        -- The warm inventory cache is already sorted by value; keep its order.
        -- Keep both expensive and cheap cards: small cards are often necessary
        -- to fill the last part of an offer without making the search unbounded.
        if #eligible > 120 then
            local trimmed = {}
            for index = 1, 80 do trimmed[#trimmed + 1] = eligible[index] end
            for index = #eligible - 39, #eligible do trimmed[#trimmed + 1] = eligible[index] end
            eligible = trimmed
        end
        availableSlots = math.min(availableSlots, maxItems - placed)

        local states = {{sum = 0, count = 0}}
        local seen = {["0:0"] = true}
        local MAX_STATES = 240
        local function pruneStates()
            table.sort(states, function(a, b)
                if math.abs(remaining - a.sum) ~= math.abs(remaining - b.sum) then
                    return math.abs(remaining - a.sum) < math.abs(remaining - b.sum)
                end
                if a.count ~= b.count then return a.count < b.count end
                return a.sum > b.sum
            end)
            while #states > MAX_STATES do
                table.remove(states)
            end
            table.clear(seen)
            for _, state in ipairs(states) do
                seen[state.count .. ":" .. tostring(math.floor(state.sum * 100 + 0.5))] = true
            end
        end

        for index, item in ipairs(eligible) do
            local stateCount = #states
            for stateIndex = 1, stateCount do
                local state = states[stateIndex]
                if state.count < availableSlots then
                    local sum = state.sum + item.Value
                    if sum <= remaining + 0.001 then
                        local picksCount = state.count + 1
                        local key = picksCount .. ":" .. tostring(math.floor(sum * 100 + 0.5))
                        if not seen[key] then
                            seen[key] = true
                            states[#states + 1] = {
                                sum = sum,
                                count = picksCount,
                                previous = state,
                                item = item
                            }
                        end
                    end
                end
            end
            if #states > MAX_STATES * 3 or (#states > MAX_STATES * 2 and index % 4 == 0) then
                pruneStates()
            end
            if index % 24 == 0 then task.wait() end
        end

        local best
        for _, state in ipairs(states) do
            if state.count > 0 and (not best or state.sum > best.sum
                or (state.sum == best.sum and state.count < best.count)) then
                best = state
            end
        end
        if not best then return nil end
        local picks = {}
        while best and best.item do
            picks[#picks + 1] = best.item
            best = best.previous
        end
        for left = 1, math.floor(#picks / 2) do
            local right = #picks - left + 1
            picks[left], picks[right] = picks[right], picks[left]
        end
        return picks
    end

    local selectionPasses = 0
    local plannedBundle, plannedMine, plannedTheir, plannedSlots
    while placed < maxItems and selectionPasses < 12 do
        selectionPasses = selectionPasses + 1
        local currentMine, currentSlots = readOfferState(yourOffer)
        local currentTheir = readOfferState(theirOffer)
        if plannedBundle and plannedMine and plannedSlots == currentSlots
            and plannedTheir and math.abs(plannedTheir - currentTheir) <= 0.5
            and currentMine < plannedMine then
            -- The slot appeared before its label/amount finished updating.
            -- Keep the just-placed item's known value in the live plan.
            currentMine = plannedMine
        end
        actualTarget = currentTheir * factor
        local remaining = actualTarget - currentMine
        if remaining <= 0.5 then break end

        if not plannedBundle or math.abs((plannedMine or -1) - currentMine) > 0.5
            or math.abs((plannedTheir or -1) - currentTheir) > 0.5 or plannedSlots ~= currentSlots then
            plannedBundle = findBestBundle(remaining, math.min(maxItems - placed, 4 - currentSlots))
            plannedMine, plannedTheir, plannedSlots = currentMine, currentTheir, currentSlots
        end
        local item = plannedBundle and table.remove(plannedBundle, 1)
        if not item then break end

        -- Recheck immediately before clicking so a changed offer cannot make
        -- the selected item exceed the updated target.
        local checkedMine, checkedSlots = readOfferState(yourOffer)
        local checkedTheir = readOfferState(theirOffer)
        if plannedBundle and plannedMine and plannedSlots == checkedSlots
            and plannedTheir and math.abs(plannedTheir - checkedTheir) <= 0.5
            and checkedMine < plannedMine then
            checkedMine = plannedMine
        end
        local checkedTarget = checkedTheir * factor
        local checkedRemaining = checkedTarget - checkedMine
        if checkedRemaining <= 0.5 then
            actualTarget = checkedTarget
            break
        end

        local retryPlan = math.abs(checkedRemaining - remaining) > 0.5 or item.Value > checkedRemaining
        if not retryPlan then
            currentMine, currentSlots = checkedMine, checkedSlots
            actualTarget = checkedTarget
            local initialFrame = item.Frame
            if not initialFrame or not CanClickItem(initialFrame) then
                failedName = item.Name
                break
            end

            local beforeValue, beforeSlots = currentMine, currentSlots
            local added, clickedFrame = false, nil
            local observedMine, observedTheir, observedSlots = beforeValue, currentTheir, beforeSlots
            for attempt = 1, 2 do
                clickedFrame = item.Frame
                if not clickedFrame or not clickedFrame.Parent or not clickedFrame.Visible then break end
                InventoryCache.ExpectedFrames[clickedFrame] = true
                ClickItem(clickedFrame, attempt == 2)
                local deadline = os.clock() + 0.65
                repeat
                    task.wait(0.08)
                    observedMine, observedSlots = readOfferState(yourOffer)
                    observedTheir = readOfferState(theirOffer)
                    if observedSlots > beforeSlots or observedMine >= beforeValue + math.max(0.5, item.Value * 0.55) then
                        added = true
                        break
                    end
                until os.clock() >= deadline
                if added then break end
            end
            if added then
                placed = placed + 1
                if clickedFrame then
                    usedFrames[clickedFrame] = true
                    InventoryCache.ConsumedFrames[#InventoryCache.ConsumedFrames + 1] = clickedFrame
                    InventoryCache.ExpectedFrames[clickedFrame] = nil
                end
                selectedItems[#selectedItems + 1] = item
                -- Keep the remaining planned items when the actual offer still
                -- matches the plan. Recompute only after another player's offer
                -- or our current slot/value state has changed.
                plannedMine = math.max(observedMine, beforeValue + item.Value)
                plannedTheir, plannedSlots = observedTheir, observedSlots
            else
                if clickedFrame then InventoryCache.ExpectedFrames[clickedFrame] = nil end
                -- The click result was ambiguous; let the background refresh
                -- reconcile the inventory once the generator has stopped.
                InventoryCache.DirtyDuringGeneration = true
                failedName = item.Name
                break
            end
        else
            plannedBundle = nil
            task.wait(0.05)
        end
    end
    Hub.WindowGui.Enabled = windowWasEnabled

    local lines = {}
    for _, item in ipairs(selectedItems) do
        lines[#lines + 1] = string.format("%s — %d", item.Name, item.Value)
    end
    local finalMine = readOfferState(yourOffer)
    local finalTheir = readOfferState(theirOffer)
    actualTarget = finalTheir * factor
    if failedName then
        setTradeStatus(string.format("Placed %d items; could not add %s. Your offer: %d / target %d. %s",
            placed, failedName, finalMine, math.floor(actualTarget), table.concat(lines, ", ")))
    elseif placed == 0 and finalMine < actualTarget then
        setTradeStatus(string.format("No available item fits the remaining target (%d). Your offer: %d / %d.",
            math.floor(actualTarget - finalMine), finalMine, math.floor(actualTarget)))
    else
        setTradeStatus(string.format("Placed %d items | You: %d | Them: %d | Target: %d | %s",
            placed, finalMine, finalTheir, math.floor(actualTarget), table.concat(lines, ", ")))
    end
end

local AutoSelectingOffer = false
local function AutoSelectOffer(mode)
    if AutoSelectingOffer then return end
    AutoSelectingOffer = true
    if HubContext.setGeneratorBusy then HubContext.setGeneratorBusy(true) end
    InventoryCache.DirtyDuringGeneration = false
    table.clear(InventoryCache.ExpectedFrames)
    table.clear(InventoryCache.ConsumedFrames)
    task.spawn(function()
        -- Finish any item cards already being cached before freezing inventory
        -- updates for this offer-generation run.
        local queueDeadline = os.clock() + 3
        while (InventoryCache.CardQueueRunning or next(InventoryCache.CardQueue))
            and os.clock() < queueDeadline do
            task.wait(0.05)
        end
        if InventoryCache.CardQueueRunning or next(InventoryCache.CardQueue) then
            AutoSelectingOffer = false
            if HubContext.setGeneratorBusy then HubContext.setGeneratorBusy(false) end
            setTradeStatus("Inventory updates are still processing. Try again in a moment.")
            return
        end
        InventoryCache.Paused = true
        local ok, err = pcall(AutoSelectOfferNow, mode)
        InventoryCache.Paused = false
        -- A normal placement only removes cards we explicitly clicked. Purge
        -- those from the warm list and avoid rescanning the whole inventory.
        if InventoryCache.DirtyDuringGeneration then
            InventoryCache.Dirty = true
        else
            for _, frame in ipairs(InventoryCache.ConsumedFrames) do
                for index = #InventoryCache.Items, 1, -1 do
                    if InventoryCache.Items[index].Frame == frame then
                        table.remove(InventoryCache.Items, index)
                    end
                end
            end
            InventoryCache.Groups = summarizePricedItems(InventoryCache.Items)
            InventoryCache.Ready = #InventoryCache.Items > 0
            InventoryCache.Dirty = false
        end
        table.clear(InventoryCache.ExpectedFrames)
        table.clear(InventoryCache.ConsumedFrames)
        pcall(function() Hub.WindowGui.Enabled = true end)
        AutoSelectingOffer = false
        if HubContext.setGeneratorBusy then HubContext.setGeneratorBusy(false) end
        if not ok then
            setTradeStatus("Offer generation stopped: " .. tostring(err))
        end
    end)
end

local tradeDisplayCache = setmetatable({}, {__mode = "k"})

local function UpdateTradeUI()
    if not hubAlive then return end
    if not Hub.WindowGui or not Hub.WindowGui.Parent then return end
    if not Hub.WindowGui.Enabled then return end
    if not Hub.Visible then return end
    if not HubContext.tradeValue or not HubContext.tradeVerdict then return end

    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    if not playerGui then return end

    local tradeGui = GetActiveTradeGui(playerGui)
    if not tradeGui then
        HubContext.tradeVerdict.Set("idle", "WAITING FOR TRADE")
        HubContext.tradeValue.SetText("You: 0  |  Them: 0")
        return
    end

    local tradeContainer = tradeGui:FindFirstChild("Container")
    local tradeMain = tradeContainer and (tradeContainer:FindFirstChild("Trade") or tradeContainer:FindFirstChild("Main") or tradeContainer)

    local isTradeActive = false
    if tradeGui and tradeContainer then
        if tradeGui:IsA("ScreenGui") then
            isTradeActive = tradeGui.Enabled and tradeContainer.Visible
        else
            isTradeActive = tradeGui.Visible and tradeContainer.Visible
        end
    end

    if not isTradeActive then
        HubContext.tradeVerdict.Set("idle", "WAITING FOR TRADE")
        HubContext.tradeValue.SetText("You: 0  |  Them: 0")
        return
    end

    local yourOffer = tradeMain and tradeMain:FindFirstChild("YourOffer") and (tradeMain.YourOffer:FindFirstChild("Container") or tradeMain.YourOffer)
    local theirOffer = tradeMain and tradeMain:FindFirstChild("TheirOffer") and (tradeMain.TheirOffer:FindFirstChild("Container") or tradeMain.TheirOffer)

    local MyTotalValue = 0
    local HisTotalValue = 0

    local function ProcessContainer(container, isMine)
        if not container then return end
        for _, itemFrame in pairs(container:GetChildren()) do
            if itemFrame:IsA("GuiObject") then
                local isSlotActive = itemFrame.Visible
                local cached = tradeDisplayCache[itemFrame]
                local itemNameContainer = cached and cached.Label
                if not itemNameContainer or not itemNameContainer.Parent
                    or not itemNameContainer:IsDescendantOf(itemFrame) then
                    itemNameContainer = itemFrame:FindFirstChild("ItemName") or itemFrame:FindFirstChild("Label")
                end
                if itemNameContainer and itemNameContainer:IsA("GuiObject") then
                    if not itemNameContainer.Visible then isSlotActive = false end
                end
                local itemNameValue = ""
                if isSlotActive and itemNameContainer then
                    local label = itemNameContainer:IsA("TextLabel") and itemNameContainer or itemNameContainer:FindFirstChild("Label")
                    if label and label:IsA("TextLabel") then
                        itemNameValue = label.Text
                    end
                end
                if isSlotActive and itemNameValue ~= "" and itemNameValue ~= "Item" then
                    local icon = cached and cached.Icon
                    if not icon or not icon.Parent or not icon:IsDescendantOf(itemFrame) then
                        local cardContainer = itemFrame:FindFirstChild("Container", true)
                        icon = cardContainer and cardContainer:FindFirstChild("Icon", true)
                        if not icon then icon = itemFrame:FindFirstChild("Icon", true) end
                    end
                    local image = icon and (icon:IsA("ImageLabel") or icon:IsA("ImageButton")) and icon.Image or nil
                    if not cached or cached.Name ~= itemNameValue or cached.Image ~= image then
                        cached = {
                            Name = itemNameValue,
                            Label = itemNameContainer,
                            Icon = icon,
                            Image = image,
                            Amount = itemFrame:FindFirstChild("Amount", true),
                            Data = GetItemData(itemNameValue, false, false, itemFrame)
                        }
                        tradeDisplayCache[itemFrame] = cached
                    else
                        cached.Label = itemNameContainer
                        cached.Icon = icon
                        if not cached.Amount or not cached.Amount:IsDescendantOf(itemFrame) then
                            cached.Amount = itemFrame:FindFirstChild("Amount", true)
                        end
                    end
                    local itemData = cached.Data
                    local overlay = cached.Overlay
                    if not overlay or not overlay.Parent or not overlay:IsDescendantOf(itemFrame) then
                        overlay = itemFrame:FindFirstChild("NovaOverlayFinal")
                    end
                    if not overlay then
                        overlay = Instance.new("TextLabel")
                        overlay.Name = "NovaOverlayFinal"
                        overlay.Size = UDim2.new(1, 0, 1, 0)
                        overlay.BackgroundTransparency = 1
                        overlay.TextTransparency = 0
                        overlay.TextSize = (TargetGuiName == "TradeGUI_Phone") and 13 or 14
                        overlay.TextWrapped = true
                        overlay.Font = FONT_BOLD
                        overlay.TextStrokeTransparency = 0.2
                        overlay.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                        overlay.ZIndex = 100
                        overlay.Parent = itemFrame
                    end
                    cached.Overlay = overlay
                    if itemData then
                        if itemData.Value and itemData.Value > 0 then
                            local count = GetItemCount(itemFrame, cached.Amount)
                            local totalItemValue = itemData.Value * count
                            local overlayText = tostring(totalItemValue) .. "\n(" .. tostring(itemData.Stability or "N/A") .. ")" .. (count > 1 and (" x" .. count) or "")
                            if overlay.Visible ~= true then overlay.Visible = true end
                            if overlay.TextTransparency ~= 0 then overlay.TextTransparency = 0 end
                            if overlay.Text ~= overlayText then overlay.Text = overlayText end
                            overlay.TextColor3 = Color3.fromRGB(255, 255, 255)
                            if isMine then MyTotalValue = MyTotalValue + totalItemValue
                            else HisTotalValue = HisTotalValue + totalItemValue end
                        else
                            if overlay.Text ~= "" then overlay.Text = "" end
                            if overlay.Visible then overlay.Visible = false end
                        end
                    else
                        if overlay.Text ~= "" then overlay.Text = "" end
                        if overlay.Visible then overlay.Visible = false end
                    end
                else
                    if cached then
                        tradeDisplayCache[itemFrame] = nil
                        local overlay = cached.Overlay or itemFrame:FindFirstChild("NovaOverlayFinal")
                        if overlay then overlay:Destroy() end
                    end
                end
            end
        end
    end

    ProcessContainer(yourOffer, true)
    ProcessContainer(theirOffer, false)

    HubContext.tradeValue.SetText("YOU: " .. tostring(MyTotalValue) .. "  |  THEM: " .. tostring(HisTotalValue))
    local difference = HisTotalValue - MyTotalValue

    if MyTotalValue == 0 and HisTotalValue == 0 then
        HubContext.tradeVerdict.Set("idle", "EMPTY TRADE")
    elseif difference > (MyTotalValue * 0.1) then
        HubContext.tradeVerdict.Set("ok", "WIN")
    elseif difference < -(MyTotalValue * 0.1) then
        HubContext.tradeVerdict.Set("err", "LOSE")
    else
        HubContext.tradeVerdict.Set("warn", "FAIR")
    end
end

-- --------------------------------------------------------------------------
-- Inventory analysis (from the reference's GetFullInventory analyzer,
-- price source adapted to this hub's Supreme Values database)
-- --------------------------------------------------------------------------
local function supremeValueFor(key)
    local candidates = Database[normalize(key)]
    if not candidates then return nil end
    if candidates.Value then return tonumber(candidates.Value) end
    for _, entry in ipairs(candidates) do
        local v = tonumber(entry.Value)
        if v and v > 0 then return v end
    end
    return nil
end

local function getFullInventoryTotals(playerName)
    local ok, response = pcall(function()
        return ReplicatedStorage.Remotes.Extras.GetFullInventory:InvokeServer(playerName)
    end)
    if not ok or type(response) ~= "table" then return nil end
    local totals = {}
    for _, category in ipairs({"Weapons", "Pets"}) do
        local owned = (response[category] or {}).Owned or {}
        for key, count in pairs(owned) do
            local countNumber = tonumber(count) or 0
            totals[key] = (totals[key] or 0) + countNumber
        end
    end
    return totals
end

local function buildInventoryReport(playerName)
    if not isDatabaseLoaded then
        return nil, "The Supreme value list is still loading. Try again shortly."
    end
    local totals = getFullInventoryTotals(playerName)
    if not totals then
        return nil, "GetFullInventory remote is unavailable or refused the request."
    end

    local rows = {}
    local totalCount, totalValue = 0, 0
    for key, count in pairs(totals) do
        local unit = supremeValueFor(key)
        if unit and unit > 0 and count and count > 0 then
            local total = unit * count
            rows[#rows + 1] = {name = key, count = count, unit = unit, total = total}
            totalCount = totalCount + count
            totalValue = totalValue + total
        end
    end
    table.sort(rows, function(a, b) return a.total > b.total end)

    if #rows == 0 then
        return "Inventory Analysis for " .. tostring(playerName) .. ":\nNo items matched Supreme Values entries."
    end

    local lines = {
        "Inventory Analysis: " .. tostring(playerName),
        string.rep("-", 36)
    }
    local shown = math.min(#rows, 40)
    for i = 1, shown do
        local row = rows[i]
        lines[#lines + 1] = string.format("%-22s x%-4d | %6d ea | %9d", string.sub(row.name, 1, 22), row.count, row.unit, row.total)
    end
    if #rows > shown then
        lines[#lines + 1] = string.format("... and %d more priced entries", #rows - shown)
    end
    lines[#lines + 1] = string.rep("-", 36)
    lines[#lines + 1] = string.format("Priced items: %d (of %d owned types)", #rows, totalCount > 0 and totalCount or 0)
    lines[#lines + 1] = string.format("Total value: %d", math.floor(totalValue))
    return table.concat(lines, "\n")
end

-- Murderer / sheriff detection: knives and guns in backpacks/characters first,
-- then the PlayerDataChanged role table (u46 in the reference).
local function findMurderer()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player:FindFirstChild("Backpack") and player.Backpack:FindFirstChild("Knife") then
            return player
        end
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("Knife") then
            return player
        end
    end
    if roleData then
        for name, data in pairs(roleData) do
            if type(data) == "table" and data.Role == "Murderer" and Players:FindFirstChild(name) then
                return Players:FindFirstChild(name)
            end
        end
    end
    -- The local player can be the murderer; report them last.
    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer and player:FindFirstChild("Backpack") and player.Backpack:FindFirstChild("Knife") then
            return player
        end
    end
    return nil
end

-- --------------------------------------------------------------------------
-- Fly coin farm (reference "Coin Farm 1"): BodyVelocity + tween to the
-- nearest uncollected coin; anchored hover when none are in range.
-- --------------------------------------------------------------------------
local flyFarm = { connection = nil, stopHook = nil }

function flyFarm.cleanup()
    if flyFarm.heartbeat then
        pcall(function() flyFarm.heartbeat:Disconnect() end)
        flyFarm.heartbeat = nil
    end
    if flyFarm.fullBagHook then
        pcall(function() flyFarm.fullBagHook:Disconnect() end)
        flyFarm.fullBagHook = nil
    end
    if flyFarm.roundEndHook then
        pcall(function() flyFarm.roundEndHook:Disconnect() end)
        flyFarm.roundEndHook = nil
    end
    if flyFarm.diedHook then
        pcall(function() flyFarm.diedHook:Disconnect() end)
        flyFarm.diedHook = nil
    end
    local root = getRoot()
    if root then
        root.Anchored = false
        local velocity = root:FindFirstChild("Nova_CoinFarmVelocity")
        if velocity then velocity:Destroy() end
    end
end

function flyFarm.nearestCoin()
    local root = getRoot()
    if not root then return nil end
    local best, bestDistance = nil, math.huge
    for _, model in ipairs(Workspace:GetChildren()) do
        if model:IsA("Model") and model:FindFirstChild("CoinContainer") then
            for _, coin in pairs(model.CoinContainer:GetChildren()) do
                if coin.Name == "Coin_Server" and not coin:FindFirstChild("CollectedCoin") and coin:FindFirstChild("TouchInterest") then
                    local distance = (root.Position - coin.Position).Magnitude
                    if distance < 100 and distance < bestDistance then
                        bestDistance = distance
                        best = coin
                    end
                end
            end
        end
    end
    return best
end

function flyFarm.start()
    flyFarm.cleanup()
    if not SharedRemotes.CoinCollected then
        notify("Auto Farm", "CoinCollected remote was not found; the fly farm cannot track progress.", "Danger", 5)
        return false
    end

    flyFarm.fullBagHook = SharedRemotes.CoinCollected.OnClientEvent:Connect(function(_, bag, maxBag)
        -- When the bag is full (bag == max) the reference stops moving.
        if bag ~= nil and maxBag ~= nil and bag == maxBag then
            flyFarm.cleanup()
        end
    end)

    if SharedRemotes.RoundEndFade then
        flyFarm.roundEndHook = SharedRemotes.RoundEndFade.OnClientEvent:Connect(function()
            flyFarm.cleanup()
        end)
    end

    local humanoid = getHumanoid()
    if humanoid then
        flyFarm.diedHook = humanoid.Died:Connect(function()
            task.wait(1)
            if AutoFarm.State ~= "running" or AutoFarm.Mode ~= "Fly" then return end
            local deadline = os.clock() + 20
            while os.clock() < deadline do
                if LocalPlayer.Character and getRoot() and LocalPlayer:FindFirstChild("PlayerGui")
                    and LocalPlayer.PlayerGui:FindFirstChild("MainGUI") then break end
                task.wait(0.5)
            end
            task.wait(1)
            teleportToMapSpawn()
        end)
    end

    flyFarm.heartbeat = RunService.Heartbeat:Connect(function()
        if AutoFarm.State ~= "running" or AutoFarm.Mode ~= "Fly" then return end
        local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
        local mainGui = playerGui and playerGui:FindFirstChild("MainGUI")
        local gameFrame = mainGui and mainGui:FindFirstChild("Game")
        local bagOk, bagVisible, fullVisible = pcall(function()
            return gameFrame.CoinBags.Visible, gameFrame.CoinBags.Container.Candy.Full.Visible
        end)
        local root = getRoot()
        if bagOk and bagVisible and not fullVisible and root then
            if not root:FindFirstChild("Nova_CoinFarmVelocity") then
                local bodyVelocity = Instance.new("BodyVelocity")
                bodyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
                bodyVelocity.Name = "Nova_CoinFarmVelocity"
                bodyVelocity.Parent = root
            end
            disableCharacterCollide()

            local coin = flyFarm.nearestCoin()
            if coin then
                root.Anchored = false
                -- Throttle tween creation: the heartbeat runs every frame, but
                -- a fresh tween is only needed when the target coin changes or
                -- the previous tween finished (prevents tween spam).
                local now = os.clock()
                if flyFarm.currentCoin ~= coin or now - (flyFarm.lastTween or 0) > 0.25 then
                    flyFarm.currentCoin = coin
                    flyFarm.lastTween = now
                    task.spawn(function()
                        local tweenRoot = getRoot()
                        if not tweenRoot then return end
                        local duration = (tweenRoot.Position - coin.Position).Magnitude / math.max(AutoFarm.Speed, 1)
                        local target = CFrame.new(coin.Position + Vector3.new(0, -1, 0))
                        local tween = TweenService:Create(tweenRoot, TweenInfo.new(math.max(duration, 0.05), Enum.EasingStyle.Linear, Enum.EasingDirection.Out), {CFrame = target})
                        tween:Play()
                        if coin and coin.Parent then
                            local touchHook
                            touchHook = coin.Touched:Connect(function()
                                task.wait(0.1)
                                if coin and coin.Parent then
                                    coin:Destroy()
                                end
                                if touchHook then touchHook:Disconnect() end
                            end)
                        end
                    end)
                end
            else
                root.Anchored = true
                flyFarm.currentCoin = nil
            end
        elseif root then
            -- Round UI not visible: hold position without driving the camera.
            local velocity = root:FindFirstChild("Nova_CoinFarmVelocity")
            if velocity then velocity:Destroy() end
        end
    end)
    return true
end

-- --------------------------------------------------------------------------
-- Safe-TP coin farm (reference "Coin Farm 2"): hide under the map on a safe
-- platform, then teleport beneath each coin and fire touch signals.
-- --------------------------------------------------------------------------
local safeFarm = { runningThread = nil, stopRequested = false }

function safeFarm.cleanup()
    safeFarm.stopRequested = true
    if safeFarm.runningThread then
        pcall(function() task.cancel(safeFarm.runningThread) end)
        safeFarm.runningThread = nil
    end
    local root = getRoot()
    if root then
        root.Anchored = false
        local velocity = root:FindFirstChild("Nova_CoinFarmVelocity")
        if velocity then velocity:Destroy() end
    end
    restoreCharacterCollide()
end

function safeFarm.start()
    if type(ExecutorEnv.firetouchinterest) ~= "function" then
        notify("Auto Farm", "Safe-TP farm requires firetouchinterest, which this executor does not provide. Use the Fly mode instead.", "Danger", 6)
        return false
    end
    if not SharedRemotes.CoinCollected then
        notify("Auto Farm", "CoinCollected remote was not found; the safe farm cannot track progress.", "Danger", 5)
        return false
    end

    safeFarm.stopRequested = false
    safeFarm.runningThread = task.spawn(function()
        teleportToSafespot()
        task.wait(0.3)
        while not safeFarm.stopRequested and AutoFarm.State == "running" and AutoFarm.Mode == "SafeTP" and not AutoFarm.bagFull do
            local map = findMapModel()
            if not map then
                task.wait(1)
                if safeFarm.stopRequested then break end
                teleportToSafespot()
            else
                -- Pick the lowest coin that still has a TouchInterest.
                local bestCoin, lowestY = nil, math.huge
                for _, coin in ipairs(map.CoinContainer:GetChildren()) do
                    if coin:FindFirstChild("TouchInterest") and coin.Position.Y < lowestY then
                        lowestY = coin.Position.Y
                        bestCoin = coin
                    end
                end
                if not bestCoin then
                    task.wait(0.5)
                    teleportToSafespot()
                    task.wait(1)
                else
                    local collectedHere = false
                    local bagHook = SharedRemotes.CoinCollected.OnClientEvent:Connect(function()
                        collectedHere = true
                    end)

                    local function hoverCoin()
                        local root = getRoot()
                        if root and bestCoin and bestCoin.Parent then
                            root.CFrame = CFrame.new(bestCoin.Position + Vector3.new(0, -6, 0))
                        end
                    end

                    hoverCoin()
                    while not safeFarm.stopRequested and AutoFarm.State == "running"
                        and not collectedHere and bestCoin and bestCoin.Parent
                        and bestCoin:FindFirstChild("TouchInterest") do
                        local root = getRoot()
                        if root then
                            pcall(function()
                                ExecutorEnv.firetouchinterest(root, bestCoin, 0)
                                ExecutorEnv.firetouchinterest(root, bestCoin, 1)
                            end)
                        end
                        disableCharacterCollide()
                        hoverCoin()
                        task.wait(0.05)
                    end

                    if bagHook then pcall(function() bagHook:Disconnect() end) end
                    if safeFarm.stopRequested or AutoFarm.State ~= "running" then break end
                    teleportToSafespot()
                    task.wait(1.5)
                end
            end
        end
        restoreCharacterCollide()
    end)
    return true
end

-- A new round resets the bag and restarts the fly farm cleanly.
-- (Placed after flyFarm/safeFarm so the closures capture the locals.)
if SharedRemotes.RoundStart then
    GlobalScope.add(SharedRemotes.RoundStart.OnClientEvent:Connect(function()
        AutoFarm.BagCoins = 0
        AutoFarm.bagFull = false
        if AutoFarm.State == "running" and AutoFarm.Mode == "Fly" then
            pcall(function() flyFarm.cleanup() end)
            task.delay(0.5, function()
                if AutoFarm.State == "running" and AutoFarm.Mode == "Fly" then
                    flyFarm.start()
                end
            end)
        end
    end))
end

-- --------------------------------------------------------------------------
-- Session clock + overlay renderer (single loop, 2 Hz)
-- --------------------------------------------------------------------------
task.spawn(function()
    local lastTick = os.clock()
    local lastRender = 0
    while hubAlive do
        local now = os.clock()
        local dt = now - lastTick
        lastTick = now
        if AutoFarm.State == "running" then
            AutoFarm.SessionTime = AutoFarm.SessionTime + dt
        end
        if now - lastRender > 0.5 then
            lastRender = now
            AutoFarm.refreshStatsUI()
        end
        task.wait(0.1)
    end
end)

function AutoFarm.start()
    if AutoFarm.State == "running" then return end
    if AutoFarm.State == "paused" then
        AutoFarm.State = "running"
        farmStatus("ok", "RUNNING (" .. AutoFarm.Mode .. ")")
        return
    end
    AutoFarm.SessionStart = os.clock()
    local started = false
    if AutoFarm.Mode == "Fly" then
        started = flyFarm.start()
    else
        started = safeFarm.start()
    end
    if started then
        AutoFarm.State = "running"
        farmStatus("ok", "RUNNING (" .. AutoFarm.Mode .. ") | Bag " .. AutoFarm.BagCoins .. "/" .. AutoFarm.BagMax)
        notify("Auto Farm", "Started in " .. AutoFarm.Mode .. " mode.", "Success", 3)
    else
        AutoFarm.State = "stopped"
        farmStatus("err", "START FAILED — see notification")
    end
end

function AutoFarm.pause()
    if AutoFarm.State ~= "running" then return end
    AutoFarm.State = "paused"
    -- Halt motion immediately; loops no-op while paused.
    local root = getRoot()
    if root then
        root.Anchored = false
        local velocity = root:FindFirstChild("Nova_CoinFarmVelocity")
        if velocity then velocity:Destroy() end
    end
    farmStatus("warn", "PAUSED | Bag " .. AutoFarm.BagCoins .. "/" .. AutoFarm.BagMax .. " | Total " .. AutoFarm.TotalCoins)
end

function AutoFarm.stop()
    AutoFarm.State = "stopped"
    flyFarm.cleanup()
    safeFarm.cleanup()
    farmStatus("idle", "STOPPED | Total coins this session: " .. AutoFarm.TotalCoins)
end

function AutoFarm.reset()
    AutoFarm.TotalCoins = 0
    AutoFarm.BagCoins = 0
    AutoFarm.SessionTime = 0
    AutoFarm.bagFull = false
    AutoFarm.refreshStatsUI()
    if AutoFarm.State == "stopped" then
        farmStatus("idle", "Stats reset.")
    end
    notify("Auto Farm", "Session statistics were reset.", "Info", 2)
end

function AutoFarm.setMode(mode)
    AutoFarm.Mode = mode == "SafeTP" and "SafeTP" or "Fly"
    -- Restart the active farm so a mode switch takes effect immediately.
    if AutoFarm.State == "running" then
        flyFarm.cleanup()
        safeFarm.cleanup()
        AutoFarm.State = "stopped"
        AutoFarm.start()
    end
end

-- Instant gun pickup (ported from the reference's pickupgun()).
function ItemsModule.pickupGun(silent)
    local map = findMapModel()
    if not map then
        ItemsModule.LastError = "No active map with a CoinContainer was found."
        itemsStatus("err", "Pickup failed: no active map.")
        if not silent then notify("Pick Up Gun", ItemsModule.LastError, "Warning", 3) end
        return false
    end
    local gunDrop = map:FindFirstChild("GunDrop")
    if not gunDrop then
        ItemsModule.LastError = "No GunDrop is present on the current map right now."
        itemsStatus("warn", "No dropped gun on the map.")
        if not silent then notify("Pick Up Gun", ItemsModule.LastError, "Warning", 3) end
        return false
    end
    if findMurderer() == LocalPlayer then
        ItemsModule.LastError = "You are the murderer; the sheriff gun cannot be picked up."
        itemsStatus("warn", "You are the murderer.")
        if not silent then notify("Pick Up Gun", ItemsModule.LastError, "Warning", 3) end
        return false
    end
    local character = getCharacter()
    if not character then
        ItemsModule.LastError = "Character is not available."
        itemsStatus("err", ItemsModule.LastError)
        return false
    end

    local pivot = character:GetPivot()
    character:MoveTo(gunDrop.Position)

    local backpack = LocalPlayer:FindFirstChild("Backpack")
    local gun = backpack and backpack:FindFirstChild("Gun")
    local started = os.clock()
    while not gun and os.clock() - started < 2 do
        backpack = LocalPlayer:FindFirstChild("Backpack")
        gun = backpack and backpack:FindFirstChild("Gun")
        task.wait()
    end

    if character.Parent then character:PivotTo(pivot) end

    if gun then
        ItemsModule.PickupCount = ItemsModule.PickupCount + 1
        ItemsModule.LastError = ""
        itemsStatus("ok", "Gun picked up (" .. ItemsModule.PickupCount .. " this session).")
        return true
    end
    ItemsModule.LastError = "Moved to the GunDrop but no Gun arrived in the Backpack."
    itemsStatus("warn", "GunDrop touched but pickup was not granted.")
    return false
end

-- Auto pickup: watch for GunDrop instances appearing on the map.
function ItemsModule.setAutoPickup(enabled)
    ItemsModule.AutoPickup = enabled and true or false
    if enabled then
        ItemsModule.gunWatcher = Workspace.DescendantAdded:Connect(function(descendant)
            if descendant.Name ~= "GunDrop" then return end

            task.wait(0.5)
            for _ = 1, 5 do
                if not ItemsModule.AutoPickup then break end
                pcall(function() ItemsModule.pickupGun(true) end)
                task.wait(0.1)

        end)

    else
        if ItemsModule.gunWatcher then
            pcall(function() ItemsModule.gunWatcher:Disconnect() end)
            ItemsModule.gunWatcher = nil
        end
        itemsStatus("idle", "Auto pick up gun disabled.")
    end
end

-- Crate opening (reference AutoOpenCrates): fires the MysteryBox remote.
local crateCatalogue = {}
local crateTypeLookup = {}
do
    local shopOk, shopResult = pcall(function()
        return require(ReplicatedStorage.Database.Sync.Shop).Weapons
    end)
    if shopOk and type(shopResult) == "table" then
        for crateName, entry in pairs(shopResult) do
            table.insert(crateCatalogue, tostring(crateName))
            if type(entry) == "table" and entry.DataType then
                crateTypeLookup[tostring(crateName)] = tostring(entry.DataType)
            end
        end
        table.sort(crateCatalogue)
    end
end

function ItemsModule.setAutoCrates(enabled)
    ItemsModule.AutoCrates = enabled and true or false
    if enabled then
        if #crateCatalogue == 0 then
            itemsStatus("err", "Database.Sync.Shop did not expose crate entries.")
            ItemsModule.AutoCrates = false
            return
        end
        if not ItemsModule.Crate then ItemsModule.Crate = crateCatalogue[1] end
        ItemsModule.crateThread = task.spawn(function()
            local mysteryBoxRemote = SharedRemotes.Remotes
                and SharedRemotes.Remotes:FindFirstChild("MysteryBox")
                and SharedRemotes.Remotes.MysteryBox:FindFirstChild("OpenBox")
            if not mysteryBoxRemote then
                itemsStatus("err", "Remotes.MysteryBox.OpenBox was not found.")
                ItemsModule.AutoCrates = false
                if HubContext.crateToggle then pcall(function() HubContext.crateToggle.Set(false) end) end
                return
            end
            itemsStatus("ok", "Auto open crates running (" .. tostring(ItemsModule.Crate) .. ").")
            while ItemsModule.AutoCrates do
                if not ItemsModule.Crate then break end
                pcall(function()
                    mysteryBoxRemote:FireServer("OpenBox", ItemsModule.Crate, ItemsModule.CrateType)
                end)
                task.wait(0.5)
            end
            itemsStatus("idle", "Auto open crates stopped.")
        end)
    else
        ItemsModule.AutoCrates = false
        if ItemsModule.crateThread then
            pcall(function() task.cancel(ItemsModule.crateThread) end)
            ItemsModule.crateThread = nil
        end
        itemsStatus("idle", "Auto open crates disabled.")
    end
end

-- Fast skin swapper (reference ShuffleWeapons).
function ItemsModule.setShuffle(enabled)
    ItemsModule.Shuffle = enabled and true or false
    if enabled then
        ItemsModule.shuffleConn = RunService.Stepped:Connect(function()

                local equip = inventoryFolder and inventoryFolder:FindFirstChild("Equip")
                if equip then
                    equip:FireServer(picked, "Weapons")
                end
            end)
        end)
        itemsStatus("ok", "Weapon shuffle enabled.")
    else
        if ItemsModule.shuffleConn then
            pcall(function() ItemsModule.shuffleConn:Disconnect() end)
            ItemsModule.shuffleConn = nil
        end
        itemsStatus("idle", "Weapon shuffle disabled.")

function Movement.setFly(enabled)
    if enabled then
        if not isAlive() then
            notify("Fly", "Spawn a character before enabling Fly.", "Warning", 3)
            movementStatus("err", "Fly failed: no living character.")
            return false
        end
        Movement.Flying = true
        Movement.flyConn = RunService.RenderStepped:Connect(function()
            local character = getCharacter()
            local humanoid = character and character:FindFirstChildOfClass("Humanoid")
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not character or not humanoid or humanoid.Health <= 0 or not root then return end

            if not root:FindFirstChild("Nova_FlyVelocity") then
                local velocity = Instance.new("BodyVelocity")
                velocity.Name = "Nova_FlyVelocity"
                velocity.MaxForce = Vector3.new(1e9, 1e9, 1e9)
                velocity.Velocity = Vector3.zero
                velocity.Parent = root
                local gyro = Instance.new("BodyGyro")
                gyro.Name = "Nova_FlyGyro"
                gyro.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
                gyro.P = 9000
                gyro.D = 500
                gyro.Parent = root
            end
            local velocity = root:FindFirstChild("Nova_FlyVelocity")
            local gyro = root:FindFirstChild("Nova_FlyGyro")
            if not velocity or not gyro then return end

            gyro.CFrame = Workspace.CurrentCamera and Workspace.CurrentCamera.CFrame or root.CFrame

            local camera = Workspace.CurrentCamera
            local move = Vector3.zero
            if camera then
                local forward = camera.CFrame.LookVector
                local right = camera.CFrame.RightVector
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + forward end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - forward end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + right end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - right end
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move = move + Vector3.new(0, 1, 0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then move = move - Vector3.new(0, 1, 0) end

            if move.Magnitude > 0 then
                velocity.Velocity = move.Unit * Movement.FlySpeed
            else
                velocity.Velocity = Vector3.zero
            end
        end)
        movementStatus("ok", "Fly enabled (WASD + Space/Ctrl).")
        return true
    else
        Movement.Flying = false
        if Movement.flyConn then
            pcall(function() Movement.flyConn:Disconnect() end)
            Movement.flyConn = nil
        end
        local character = getCharacter()
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if root then
            local velocity = root:FindFirstChild("Nova_FlyVelocity")
            if velocity then velocity:Destroy() end
            local gyro = root:FindFirstChild("Nova_FlyGyro")
            if gyro then gyro:Destroy() end
        end
        movementStatus("idle", "Fly disabled.")
        return true
    end
end

local ESP_ROLE_COLORS = {
    Murderer = Color3.fromRGB(255, 0, 0),
    Sheriff = Color3.fromRGB(0, 150, 255),
    Innocent = Color3.fromRGB(0, 255, 0),
    Unknown = Color3.fromRGB(255, 255, 255),

-- Assigned to the forward declaration in [S7] so PlayerDataChanged can
-- trigger a refresh without a circular dependency.
VisualsRefreshPlayerESP = function()
    if Visuals.PlayerESP then Visuals.refreshPlayerESP() end

function Visuals.roleColorFor(player)
    if player == findMurderer() then return ESP_ROLE_COLORS.Murderer end
    if player == findSheriff() then return ESP_ROLE_COLORS.Sheriff end
    local role = roleOf(player)
    if role == "Murderer" then return ESP_ROLE_COLORS.Murderer end
    if role == "Sheriff" then return ESP_ROLE_COLORS.Sheriff end
    if role == "Innocent" or role == "None" then return ESP_ROLE_COLORS.Innocent end
    return ESP_ROLE_COLORS.Unknown
end

function Visuals.refreshPlayerESP()
    -- Remove highlights whose adornee disappeared, then (re)create one per
    -- living character. Recreating keeps respawns covered without a
    -- CharacterAdded listener per player.
    local livingCharacters = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player.Character and player.Character:IsDescendantOf(Workspace) then
            livingCharacters[player.Character] = player
        end
    end
    for _, child in ipairs(Workspace:GetChildren()) do
        if child.Name == "NovaPlayerESP" and child:IsA("Highlight") then
            if not child.Adornee or not livingCharacters[child.Adornee] then
                child:Destroy()
            end
        end
    end
    for character, player in pairs(livingCharacters) do
        local existing = character:FindFirstChild("NovaPlayerESP")
        if not existing then
            existing = Instance.new("Highlight")
            existing.Name = "NovaPlayerESP"
            existing.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            existing.Adornee = character
            existing.OutlineTransparency = 0
            existing.FillTransparency = Visuals.PlayerFillTransparency
            existing.Parent = Workspace
        end
        local color = Visuals.roleColorFor(player)
        existing.FillColor = color
        existing.OutlineColor = color
    end
end

function Visuals.setPlayerESP(enabled)
    Visuals.PlayerESP = enabled and true or false
    -- Generation counter: an older refresh loop stops itself when a newer
    -- loop takes over, so rapid toggling can never stack loops.
    Visuals.espGeneration = (Visuals.espGeneration or 0) + 1
    local generation = Visuals.espGeneration
    if enabled then
        Visuals.refreshPlayerESP()
        Visuals.playerLoop = task.spawn(function()
            while Visuals.PlayerESP and Visuals.espGeneration == generation and hubAlive do
                pcall(Visuals.refreshPlayerESP)
                task.wait(2)
            end
        end)
        visualsStatus("ok", "Player ESP active (red murderer / blue sheriff / green innocent).")
    else
        for _, child in ipairs(Workspace:GetChildren()) do
            if child.Name == "NovaPlayerESP" then child:Destroy() end
        end
        visualsStatus("idle", "Player ESP removed.")
    end
end

function Visuals.setFillTransparency(value)
    Visuals.PlayerFillTransparency = value

local function escapePattern(text)

-- One shot at the detected murderer with the reference's prediction math.
function Combat.shootMurderer()
    if not LocalPlayer.Character then
        combatStatus("err", "No character.")
        return false
    end
    local gun = LocalPlayer.Character:FindFirstChild("Gun") or (LocalPlayer:FindFirstChild("Backpack") and LocalPlayer.Backpack:FindFirstChild("Gun"))
    if not gun then
        combatStatus("err", "You do not have the gun.")
        notify("Shoot", "You need the sheriff gun to shoot.", "Warning", 3)
        return false
    end
    local humanoid = getHumanoid()
    if humanoid and gun.Parent == LocalPlayer.Backpack then
        humanoid:EquipTool(gun)
    end
    local murderer = findMurderer()
    if not murderer or not murderer.Character then
        combatStatus("warn", "Murderer not detected yet.")
        return false
    end
    local upperTorso = murderer.Character:FindFirstChild("UpperTorso")
    local murdererHumanoid = murderer.Character:FindFirstChildOfClass("Humanoid")
    if not upperTorso or not murdererHumanoid then
        combatStatus("warn", "Murderer rig has no UpperTorso.")
        return false
    end
    local moveDirection = murdererHumanoid.MoveDirection
    local predicted = (upperTorso.Position + upperTorso.AssemblyLinearVelocity * Vector3.new(0, 0.5, 0) * 0.14 + moveDirection * 2.8)
    pcall(function()
        local equippedGun = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Gun")
        if equippedGun then
            equippedGun.KnifeLocal.CreateBeam.RemoteFunction:InvokeServer(1, predicted, "AH2")
        end
    end)
    combatStatus("ok", "Shot fired at " .. murderer.Name .. ".")
    return true
end

function Combat.setAutoShoot(enabled)
    Combat.AutoShoot = enabled and true or false
    if enabled then
        Combat.shootConn = RunService.RenderStepped:Connect(function()
            if not Combat.AutoShoot then return end
            if findSheriff() ~= LocalPlayer then return end
            local murderer = findMurderer()
            if not murderer or not murderer.Character then return end
            local murdererRoot = murderer.Character:FindFirstChild("HumanoidRootPart")
            local myRoot = getRoot()
            if not murdererRoot or not myRoot then return end

            local direction = murdererRoot.Position - myRoot.Position
            local raycastParams = RaycastParams.new()
            raycastParams.FilterType = Enum.RaycastFilterType.Exclude
            raycastParams.FilterDescendantsInstances = { LocalPlayer.Character }
            local hit = Workspace:Raycast(myRoot.Position, direction, raycastParams)

            if not hit or hit.Instance.Parent == murderer.Character then
                Combat.shootMurderer()
            end
        end)
        combatStatus("ok", "Auto shoot armed (fires when you are sheriff).")
    else
        Combat.AutoShoot = false
        if Combat.shootConn then
            pcall(function() Combat.shootConn:Disconnect() end)
            Combat.shootConn = nil
        end
        combatStatus("idle", "Auto shoot disabled.")
    end
end

-- Reference "Kill Everyone as Murderer": equips the knife, pulls every player
-- in front of the local character, then fires the Stab remote.
function Combat.killEveryone()
    local character = LocalPlayer.Character
    if not character then return false end
    local equipped = character:FindFirstChild("Knife")
    if not equipped then
        local backpack = LocalPlayer:FindFirstChild("Backpack")
        local knife = backpack and backpack:FindFirstChild("Knife")
        local humanoid = getHumanoid()
        if knife and humanoid then
            humanoid:EquipTool(knife)
            equipped = character:FindFirstChild("Knife")
        end
    end
    if not equipped then
        combatStatus("err", "Knife not found; you must be the murderer.")
        notify("Kill Everyone", "You do not have the knife.", "Warning", 3)
        return false
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local victimRoot = player.Character:FindFirstChild("HumanoidRootPart")
            local myRoot = getRoot()
            if victimRoot and myRoot then
                victimRoot.Anchored = true
                victimRoot.CFrame = myRoot.CFrame + myRoot.CFrame.LookVector * 1
            end
        end
    end
    pcall(function()
        LocalPlayer.Character.Knife.Stab:FireServer("Slash")
    end)
    combatStatus("warn", "Kill-everyone executed (murderer round only).")
    return true
end

-- Fling implementation (reference v189): BodyVelocity + rapid CFrame cycling
-- around the victim, then restore own position. Runs only on explicit action.
function Combat.flingPlayer(playerName)
    local target
    if playerName == "all" or playerName == "others" then
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                pcall(Combat.flingSingle, player)
            end
        end
        combatStatus("warn", "Fling all executed.")
        return true
    end
    if playerName == "random" then
        local candidates = {}
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then table.insert(candidates, player) end
        end
        if #candidates == 0 then
            combatStatus("warn", "No other players to fling.")
            return false
        end
        target = candidates[math.random(1, #candidates)]
    else
        local wanted = string.lower(tostring(playerName or ""))
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer then
                if string.find(string.lower(player.Name), "^" .. escapePattern(wanted))
                    or string.find(string.lower(player.DisplayName), "^" .. escapePattern(wanted)) then
                    target = player
                    break
                end
            end
        end
    end
    if not target then
        combatStatus("warn", "Player '" .. tostring(playerName) .. "' was not found.")
        return false
    end
    return Combat.flingSingle(target)
end

function Combat.flingSingle(targetPlayer)
    local character = LocalPlayer.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    local rootPart = humanoid and humanoid.RootPart
    if not character or not humanoid or not rootPart then return false end

    local targetCharacter = targetPlayer.Character
    if not targetCharacter then return false end
    local targetHumanoid = targetCharacter:FindFirstChildOfClass("Humanoid")
    local targetRoot = targetHumanoid and targetHumanoid.RootPart
    local targetHead = targetCharacter:FindFirstChild("Head")
    if not targetRoot then return false end

    if rootPart.Velocity.Magnitude < 50 then
        Combat.OldPos = rootPart.CFrame
    end
    if targetHumanoid and targetHumanoid.Sit then return false end

    if targetHead then
        Workspace.CurrentCamera.CameraSubject = targetHead
    elseif targetHumanoid and targetRoot then
        Workspace.CurrentCamera.CameraSubject = targetHumanoid
    end
    if not targetCharacter:FindFirstChildWhichIsA("BasePart") then return false end

    local function applyCFrame(targetPart, offset, rotation)
        rootPart.CFrame = CFrame.new(targetPart.Position) * offset * rotation
        pcall(function()
            character:SetPrimaryPartCFrame(CFrame.new(targetPart.Position) * offset * rotation)
        end)
        rootPart.Velocity = Vector3.new(90000000, 900000000, 90000000)
        rootPart.RotVelocity = Vector3.new(900000000, 900000000, 900000000)
    end

    local started = os.clock()
    local spin = 0

    local bodyVelocity = Instance.new("BodyVelocity")
    bodyVelocity.Name = "Nova_FlingVel"
    bodyVelocity.Parent = rootPart
    bodyVelocity.Velocity = Vector3.new(900000000, 900000000, 900000000)
    bodyVelocity.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

    pcall(function()
        Workspace.FallenPartsDestroyHeight = (0 / 0)
    end)

    while rootPart and targetHumanoid and targetRoot.Parent do
        if targetRoot.Velocity.Magnitude < 50 then
            spin = spin + 100
            applyCFrame(targetRoot, CFrame.new(0, 1.5, 0) + targetHumanoid.MoveDirection * targetRoot.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(spin), 0, 0))
            task.wait()
            applyCFrame(targetRoot, CFrame.new(0, -1.5, 0) + targetHumanoid.MoveDirection * targetRoot.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(spin), 0, 0))
            task.wait()
        else
            applyCFrame(targetRoot, CFrame.new(0, 1.5, targetHumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
            task.wait()
            applyCFrame(targetRoot, CFrame.new(0, -1.5, -targetHumanoid.WalkSpeed), CFrame.Angles(0, 0, 0))
            task.wait()
            applyCFrame(targetRoot, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0))
            task.wait()
            applyCFrame(targetRoot, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
            task.wait()
        end

        if targetRoot.Velocity.Magnitude > 500 or targetRoot.Parent ~= targetCharacter
            or targetPlayer.Parent ~= Players or targetHumanoid.Sit
            or humanoid.Health <= 0 or started + 2 < os.clock() then
            break
        end
    end

    if bodyVelocity and bodyVelocity.Parent then
        bodyVelocity:Destroy()
    end
    humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)

    if Combat.OldPos and humanoid.Health > 0 then
        rootPart.CFrame = Combat.OldPos
    end
    if Workspace.CurrentCamera and getHumanoid() then
        Workspace.CurrentCamera.CameraSubject = getHumanoid()
    end
    combatStatus("warn", "Fling executed on " .. targetPlayer.Name .. ".")
    return true
end

function Combat.setLoopFling(enabled)
    Combat.LoopFling = enabled and true or false
    if enabled then
        Combat.loopThread = task.spawn(function()
            while Combat.LoopFling and hubAlive do
                for _, player in ipairs(Players:GetPlayers()) do
                    if not Combat.LoopFling then break end
                    if player ~= LocalPlayer then
                        pcall(function() Combat.flingSingle(player) end)
                        task.wait(0.1)
                    end
                end
                task.wait(2)
            end
        end)
        combatStatus("warn", "Loop fling all players is running.")
    else
        Combat.LoopFling = false
        combatStatus("idle", "Loop fling stopped.")
    end
end

function Combat.setAntiFling(enabled)
    Combat.AntiFling = enabled and true or false
    if enabled then
        Combat.antiFlingConn = RunService.Heartbeat:Connect(function()
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    for _, descendant in ipairs(player.Character:GetDescendants()) do
                        if descendant:IsA("BasePart") and not descendant.Anchored and descendant.Velocity.Magnitude > 100 then
                            descendant.Velocity = Vector3.zero
                            descendant.RotVelocity = Vector3.zero
                            descendant.Anchored = true
                            task.delay(0.5, function()
                                if descendant and descendant.Parent and descendant.Anchored then
                                    descendant.Anchored = false
                                end
                            end)
                        end
                    end
                end
            end
        end)
        combatStatus("ok", "Anti-fling shield active.")
    else
        if Combat.antiFlingConn then
            pcall(function() Combat.antiFlingConn:Disconnect() end)
            Combat.antiFlingConn = nil
        end
        combatStatus("idle", "Anti-fling disabled.")
    end
end

-- Anti-AFK (reference LocalPlayer.Idled behavior).
local antiAfkEnabled = true
local antiAfkConn = nil
function setAntiAfk(enabled)
    if enabled then
        if not antiAfkConn then
            antiAfkConn = LocalPlayer.Idled:Connect(function()
                pcall(function()
                    VirtualUser:ClickButton2(Vector2.new())
                end)
            end)
        end
    else
        if antiAfkConn then
            pcall(function() antiAfkConn:Disconnect() end)
            antiAfkConn = nil
        end
    end
    antiAfkEnabled = enabled and true or false
end
setAntiAfk(true)

-- --------------------------------------------------------------------------
-- Config persistence (executor filesystem only)
-- --------------------------------------------------------------------------
function saveHubConfig()
    if not ExecutorEnv.hasFilesystem then
        notify("Config", "This executor has no writefile/isfile API.", "Warning", 3)
        return
    end
    local config = {
        Layout = Hub.Layout,
        Accent = Hub.AccentName or "Nova Blue",
        ToggleKey = Hub.ToggleKey and Hub.ToggleKey.Name or "RightControl",
        AntiAfk = antiAfkEnabled and true or false,
        FarmMode = AutoFarm.Mode,
        FarmSpeed = AutoFarm.Speed,
        FarmOverlay = AutoFarm.OverlayEnabled,
        AutoPickup = ItemsModule.AutoPickup,
        AutoCrates = false, -- never auto-resume crate spam on load
        CratePick = ItemsModule.Crate,
        ShuffleWeapons = ItemsModule.Shuffle,
        Fly = Movement.Flying,
        FlySpeed = Movement.FlySpeed,
        Noclip = Movement.Noclip,
        EnforceStats = Movement.EnforceStats,
        WalkSpeed = Movement.WalkSpeed,
        JumpPower = Movement.JumpPower,
        ESPPlayers = Visuals.PlayerESP,
        ESPFill = Visuals.PlayerFillTransparency,
        ESPTraps = Visuals.TrapESP,
        ESPGun = Visuals.GunESP,
        RemoveDoF = Visuals.RemoveDoF,
        AutoShoot = Combat.AutoShoot,
        LoopFling = false,
        AntiFling = Combat.AntiFling,
    }
    local ok, err = pcall(function()
        ExecutorEnv.writefile(CONFIG_FILE, HttpService:JSONEncode(config))
    end)
    if ok then
        notify("Config", "Configuration saved.", "Success", 3)
    else
        notify("Config", "Save failed: " .. tostring(err), "Danger", 4)
    end
end

function loadHubConfig()
    if not ExecutorEnv.hasFilesystem then
        notify("Config", "This executor has no readfile/isfile API.", "Warning", 3)
        return
    end
    local existsOk, exists = pcall(ExecutorEnv.isfile, CONFIG_FILE)
    if not existsOk or not exists then
        notify("Config", "No saved configuration found.", "Info", 3)
        return
    end
    local readOk, contents = pcall(ExecutorEnv.readfile, CONFIG_FILE)
    if not readOk or type(contents) ~= "string" then
        notify("Config", "Could not read the config file.", "Danger", 3)
        return
    end
    local decodeOk, config = pcall(function() return HttpService:JSONDecode(contents) end)
    if not decodeOk or type(config) ~= "table" then
        notify("Config", "Config file is corrupted.", "Danger", 3)
        return
    end

    -- Interface
    if config.Layout then Hub.applyLayout(tostring(config.Layout)) end
    if config.ToggleKey then
        local key = Enum.KeyCode[tostring(config.ToggleKey)]
        if key then
            Hub.ToggleKey = key
            pcall(function() keybindHandle.Set(key) end)
        end
    end
    -- Features (applied through the same setters the toggles use)
    if config.FarmMode then AutoFarm.setMode(config.FarmMode == "SafeTP" and "SafeTP" or "Fly") end
    if tonumber(config.FarmSpeed) then AutoFarm.Speed = tonumber(config.FarmSpeed) end
    if config.AutoPickup then ItemsModule.setAutoPickup(true) end
    if config.ShuffleWeapons then ItemsModule.setShuffle(true) end
    if tonumber(config.FlySpeed) then Movement.FlySpeed = tonumber(config.FlySpeed) end
    if config.Noclip then Movement.setNoclip(true) end
    if config.EnforceStats then
        if tonumber(config.WalkSpeed) then Movement.WalkSpeed = tonumber(config.WalkSpeed) end
        if tonumber(config.JumpPower) then Movement.JumpPower = tonumber(config.JumpPower) end
        Movement.setEnforceStats(true)
    end
    if tonumber(config.ESPFill) then Visuals.setFillTransparency(tonumber(config.ESPFill)) end
    if config.ESPTraps then Visuals.setTrapESP(true) end
    if config.ESPGun then Visuals.setGunESP(true) end
    if config.RemoveDoF then Visuals.setRemoveDoF(true) end
    if config.AntiFling then Combat.setAntiFling(true) end
    if config.AntiAfk ~= nil then setAntiAfk(config.AntiAfk and true or false) end

    notify("Config", "Configuration loaded. Re-enable toggles as needed — stored feature states were applied.", "Success", 4)
end

local featureEntries = {
    {"Auto Farm", function() return AutoFarm.State ~= "stopped" end},
    {"Fly", function() return Movement.Flying end},

    {"Loop Fling", function() return Combat.LoopFling end},
    {"Anti-Fling", function() return Combat.AntiFling end},
}

GlobalScope.add(task.spawn(function()
    local lastCount = -1
    while hubAlive do
        -- Values status
        pcall(function()
            if isDatabaseLoaded then
                if lastCount < 0 then
                    local count = 0
                    for _ in pairs(Database) do count = count + 1 end
                    lastCount = count
                end
                homeValueStatus.Set("ok", "Values loaded: " .. lastCount .. " entries")
            else
                homeValueStatus.Set("busy", ValuesStatusText)
            end
        end)
        -- Inventory cache status
        pcall(function()
            if InventoryCache.Scanned and InventoryCache.DatabaseReady and InventoryCache.Ready then
                homeInventoryStatus.Set("ok", "Inventory cache: " .. #InventoryCache.Items .. " priced items (" .. tostring(Hub.TargetGuiName) .. ")")
            elseif InventoryCache.Scanned and not InventoryCache.DatabaseReady then
                homeInventoryStatus.Set("warn", "Inventory cache: " .. InventoryCache.Recognized .. " cards waiting for values")
            else
                homeInventoryStatus.Set("idle", "Inventory cache: open the trade screen to populate")
            end
        end)
        -- Feature list
        pcall(function()
            local active = {}
            for _, entry in ipairs(featureEntries) do
                local ok, isOn = pcall(entry[2])
                if ok and isOn then active[#active + 1] = entry[1] end
            end
            homeFeaturesText.SetContent(#active > 0 and table.concat(active, ", ") or "None")
        end)
        task.wait(2)
    end
end))

-- --------------------------------------------------------------------------
-- Unload
-- --------------------------------------------------------------------------
function unloadHub()
    if not hubAlive then return end
    hubAlive = false

    -- Stop every feature through its own cleanup path.
    pcall(function() AutoFarm.stop() end)
    pcall(function() AutoFarm.setOverlay(false) end)
    pcall(function() ItemsModule.setAutoPickup(false) end)
    pcall(function() ItemsModule.setAutoCrates(false) end)
    pcall(function() ItemsModule.setShuffle(false) end)
    pcall(function() Movement.setFly(false) end)
    pcall(function() Movement.setNoclip(false) end)
    pcall(function() Movement.setEnforceStats(false) end)
    pcall(function() Visuals.setPlayerESP(false) end)
    pcall(function() Visuals.setTrapESP(false) end)
    pcall(function() Visuals.setGunESP(false) end)
    pcall(function() Visuals.setRemoveDoF(false) end)
    pcall(function() Combat.setAutoShoot(false) end)
    pcall(function() Combat.setLoopFling(false) end)
    pcall(function() Combat.setAntiFling(false) end)
    pcall(destroySafespotPlatform)

    GlobalScope.disconnectAll()

    pcall(function() Hub.WindowGui:Destroy() end)
    pcall(function() if NotifyGui then NotifyGui:Destroy() end end)
    pcall(function() if statsOverlayGui then statsOverlayGui:Destroy() end end)
    genv.NovaHubLoaded = nil
    print("[NovaHub] unloaded")
end

genv.NovaHubLoaded = true

-- --------------------------------------------------------------------------
-- Show the hub once the intro finishes
-- --------------------------------------------------------------------------
task.spawn(function()
    -- Wait for the intro, but never forever: the 20s cap guarantees the hub
    -- still appears even if the intro thread died or its event was missed.
    local waited = 0
    while not IntroFinished and hubAlive and waited < 20 do
        task.wait(0.25)
        waited = waited + 0.25
    end
    if not hubAlive then return end
    pcall(function() IntroFinishedEvent:Destroy() end)
    local showOk, showErr = pcall(function()
        Hub.applyLayout("PC")
        TargetGuiName = Hub.TargetGuiName
        Hub.Root.Visible = true
        Hub.Visible = true
        homeTab.select()
    end)
    if not showOk then
        warn("[NovaHub] hub show failed: " .. tostring(showErr))
    end
    task.delay(0.4, function()
        pcall(function()
            notify("Nova Hub", "Loaded. Toggle the UI with " .. Hub.ToggleKey.Name .. ".", "Success", 5)
        end)
    end)
end)
